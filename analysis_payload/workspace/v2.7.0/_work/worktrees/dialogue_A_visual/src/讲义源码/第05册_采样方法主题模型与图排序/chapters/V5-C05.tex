\phantomsection\label{struct:V5-C05-CH06}
\textbf{问题提出。}设每次观测只落入$K$个互斥类别之一。若只记录类别序列，概率是逐次类别概率的乘积；若压缩为计数向量，还必须乘上产生相同计数的序列条数。贝叶斯推断要寻找一个定义在单纯形上的先验，使多项似然乘入以后仍属于同一分布族。本章将说明这个先验是狄利克雷分布，并进一步给出证据与预测，而不是只停留在未归一化后验核。

\phantomsection\label{struct:V5-C05-CH07}
% KNOWLEDGE-PLAN: KN-V5-C34-DEFINITION-002 L1
\paragraph{读前自检：类别分布与多项分布。}类别序列概率为$\prod_t\theta_{y_t}$，计数向量概率还含多项系数；请将同一计数的序列数$N!/\prod_i n_i!$乘入，核对两种观测对象的支持与概率不同。\par

\begin{definition}[类别分布与多项分布]\label{def:V5-C05-multinomial}
给定$\boldsymbol\theta\in\Delta_{K-1}$，若
\[
\Pr(Y=i\mid\boldsymbol\theta)=\theta_i,\qquad i=1,\ldots,K,
\]
则称$Y$服从以$\boldsymbol\theta$为参数的类别分布。令$Y_1,\ldots,Y_N$在给定$\boldsymbol\theta$后条件独立同分布，并定义$n_i=\sum_{t=1}^N\mathbf 1\{Y_t=i\}$。计数向量$\boldsymbol n$的支持为
\[
\mathcal C_{N,K}
=\{\boldsymbol n\in\mathbb N_0^K:\textstyle\sum_{i=1}^K n_i=N\},
\]
其概率质量函数为
\begin{equation}\label{eq:V5-C05-multinomial-pmf}
\Pr(\boldsymbol n\mid N,\boldsymbol\theta)
=\frac{N!}{\prod_{i=1}^K n_i!}
\prod_{i=1}^K\theta_i^{n_i},
\qquad \boldsymbol n\in\mathcal C_{N,K}.
\end{equation}
约定$0^0=1$只用于表示计数为零时该因子不参与乘积。
\end{definition}
\index{多项分布}
\index{类别分布}

对一条确定的有序序列$\boldsymbol y=(y_1,\ldots,y_N)$，
\begin{equation}\label{eq:V5-C05-sequence-likelihood}
\Pr(\boldsymbol y\mid\boldsymbol\theta)
=\prod_{t=1}^N\theta_{y_t}
=\prod_{i=1}^K\theta_i^{n_i}.
\end{equation}
式\eqref{eq:V5-C05-multinomial-pmf}中的组合系数计算同一计数对应的序列数。它与$\boldsymbol\theta$无关，所以在推导后验参数核时可以消去；在计算计数向量的概率或边际证据时不能遗漏。图\ref{fig:V5-C05-multinomial-counts}把“序列压缩为计数”与“计数概率重新乘组合系数”画成两条不同路径。

\phantomsection\label{struct:V5-C05-CH08}
\textbf{模型。}本章的基本贝叶斯模型是
\begin{equation}\label{eq:V5-C05-bayes-model}
\boldsymbol\theta\sim\operatorname{Dir}(\boldsymbol\alpha),
\qquad
Y_t\mid\boldsymbol\theta\overset{\mathrm{iid}}{\sim}
\operatorname{Categorical}(\boldsymbol\theta),
\quad t=1,\ldots,N.
\end{equation}
模型对象包括类别集合、$K$维参数向量、参考单纯形、严格正超参数$\boldsymbol\alpha$、观测序列或经核验的计数向量。若只收到计数，必须同时收到$N$或核验$N=\sum_i n_i$；不能把概率参数$\boldsymbol\theta$与随机计数$\boldsymbol n$混为一类向量。

\phantomsection\label{struct:V5-C05-CH09}
\textbf{学习策略。}先把多项似然写成各$\theta_i$的幂，再选择具有同样乘积幂结构的先验。观测到$n_i$次类别$i$后，指数由$\alpha_i-1$增加到$\alpha_i+n_i-1$。这条“指数相加”规则说明分布族闭合，但完整推断还要核验三件事：先验是否可正规化，后验归一化常数是什么，以及预测与证据中哪些组合系数需要保留。

\SLTeachSection{Gamma函数与Beta分布}{sec:V5-C05-S02}

\phantomsection\label{struct:V5-C05-CH11}
% CORE-DERIVATION-PLAN: KN-V5-C34-DERIVATION-001
\SLKnowledgeBridge{通过变量变换把 Gamma 积分分解为总量和比例，并推广到单纯形。}{正参数 $a,b$、独立 Gamma 变量及变换 $s=x+y$、$t=x/(x+y)$。}{统一采用 shape--rate 参数化；明确变换区域并计入 Jacobian $s$。}{先把 $\Gamma(a)\Gamma(b)$ 写成第一象限二重积分，再作 $(x,y)=(st,s(1-t))$ 变换。}

\begin{derivationgoalbox}
从Gamma积分建立Beta--Gamma恒等式，再把它推广为多元Beta函数；随后利用独立Gamma变量的和与比例证明狄利克雷密度归一化，并为矩、共轭和后验预测提供统一计算工具。
\end{derivationgoalbox}

\begin{derivationprepbox}
以下Gamma分布统一采用shape--rate参数化：
\[
g(x\mid a,b)=\frac{b^a}{\Gamma(a)}x^{a-1}e^{-bx},
\qquad x>0,\ a,b>0.
\]
rate是指数项中的$b$；若把scale写成$b$会导致后续和分布与数值计算全部错位。变量变换证明同时记录取值域与Jacobian。
\end{derivationprepbox}

对$a>0$，
\begin{equation}\label{eq:V5-C05-gamma-definition}
\Gamma(a)=\int_0^\infty x^{a-1}e^{-x}\,\mathrm dx.
\end{equation}
分部积分给出$\Gamma(a+1)=a\Gamma(a)$，并由$\Gamma(1)=1$得到$\Gamma(n)=(n-1)!$。二元Beta函数定义为
\begin{equation}\label{eq:V5-C05-beta-definition}
B(a,b)=\int_0^1 t^{a-1}(1-t)^{b-1}\,\mathrm dt,
\qquad a,b>0.
\end{equation}

\textbf{步骤1：写成二维Gamma积分。}考虑$\Gamma(a)\Gamma(b)$并把两个正半轴积分合并为正象限积分。\par
\textbf{步骤2：改用总量与比例。}令$x=st$、$y=s(1-t)$，其中$s>0$、$0<t<1$，Jacobian绝对值为$s$。\par
\textbf{步骤3：分离两个积分。}被积函数化为$s^{a+b-1}e^{-s}t^{a-1}(1-t)^{b-1}$，从而总量积分和比例积分分别成为$\Gamma(a+b)$与$B(a,b)$。

\paragraph{读前自检：Beta--Gamma恒等式。}把Beta--Gamma恒等式放在“对$a,b>0$”下考察：把$B(a,b)$展开一次并检查其类型或边界，并检查展开后的量满足定义中的域、维数或符号限制。\par

\begin{theorem}[Beta--Gamma恒等式]\label{thm:V5-C05-beta-gamma}
对$a,b>0$，
\begin{equation}\label{eq:V5-C05-beta-gamma}
B(a,b)=\frac{\Gamma(a)\Gamma(b)}{\Gamma(a+b)}.
\end{equation}
\end{theorem}

\phantomsection\label{struct:V5-C05-CH13}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-001 L1
\paragraph{读前自检：证明：Beta--Gamma恒等式。}Beta--Gamma恒等式要求先确认“由Tonelli定理”；请随后把$\Gamma(a)\Gamma(b)$用到的关键定义展开并完成下一步变形，用变形没有增加前提且得到证明所述结论验收。\par

\begin{proof}
由Tonelli定理，非负被积函数允许把两个Gamma积分写成
\[
\Gamma(a)\Gamma(b)
=\int_0^\infty\!\!\int_0^\infty
x^{a-1}y^{b-1}e^{-(x+y)}\,\mathrm dx\,\mathrm dy.
\]
在正象限使用$x=st$、$y=s(1-t)$。变换把区域一一映到
$s>0$、$0<t<1$，且
\[
\left|\frac{\partial(x,y)}{\partial(s,t)}\right|=s.
\]
因此
\begin{align*}
\Gamma(a)\Gamma(b)
&=\int_0^\infty\!\!\int_0^1
s^{a+b-1}e^{-s}t^{a-1}(1-t)^{b-1}
\,\mathrm dt\,\mathrm ds\\
&=\Gamma(a+b)B(a,b).
\end{align*}
由于$\Gamma(a+b)>0$，两边除以该量得到式\eqref{eq:V5-C05-beta-gamma}。
\end{proof}

Beta分布的密度为
\[
f(t\mid a,b)=\frac{t^{a-1}(1-t)^{b-1}}{B(a,b)},
\qquad 0<t<1.
\]
多元Beta函数把式\eqref{eq:V5-C05-beta-gamma}推广为
\begin{equation}\label{eq:V5-C05-multivariate-beta}
B(\boldsymbol\alpha)
=\frac{\prod_{i=1}^K\Gamma(\alpha_i)}
{\Gamma(\alpha_0)},
\qquad \alpha_0=\sum_{i=1}^K\alpha_i.
\end{equation}

\paragraph{读前自检：狄利克雷分布。}若$G_i\sim\operatorname{Gamma}(\alpha_i,1)$独立并令$S=\sum_iG_i$、$\Theta_i=G_i/S$；请核对$\Theta$落在单纯形，并由Gamma归一化表示确认其为$\operatorname{Dir}(\alpha)$。\par

\begin{definition}[狄利克雷分布]\label{def:V5-C05-dirichlet}
设$K\ge2$、$\alpha_i>0$，并令
\[
\Delta_{K-1}^{\circ}
=\{\boldsymbol\theta\in(0,1)^K:
\textstyle\sum_{i=1}^K\theta_i=1\}.
\]
若$\boldsymbol\Theta$在单纯形上相对于坐标
$(\theta_1,\ldots,\theta_{K-1})$的$(K-1)$维Lebesgue测度具有密度
\begin{equation}\label{eq:V5-C05-dirichlet-density}
p(\boldsymbol\theta\mid\boldsymbol\alpha)
=\frac1{B(\boldsymbol\alpha)}
\prod_{i=1}^K\theta_i^{\alpha_i-1},
\qquad
\theta_K=1-\sum_{i=1}^{K-1}\theta_i,
\end{equation}
则记$\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha)$。边界按其极限行为理解；密度不是相对于整个$\mathbb R^K$的$K$维Lebesgue测度定义的。
\end{definition}
\index{狄利克雷分布!定义}
\index{单纯形}

当$K=2$时，$\Theta_2=1-\Theta_1$，式\eqref{eq:V5-C05-dirichlet-density}退化为
$\Theta_1\sim\operatorname{Beta}(\alpha_1,\alpha_2)$。若某个$\alpha_i<1$，密度靠近相应面时可以趋于无穷，但只要所有$\alpha_i>0$仍可积；把$\alpha_i=0$代入密度不再给出可正规化的狄利克雷分布。

\begin{theorem}[Gamma归一化构造]\label{thm:V5-C05-gamma-normalization}
设$G_i$彼此独立，且
\[
G_i\sim\operatorname{Gamma}(\alpha_i,b),
\qquad i=1,\ldots,K,
\]
其中采用共同rate $b>0$。令$S=\sum_iG_i$、$\Theta_i=G_i/S$。则
\[
\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha),
\qquad
S\sim\operatorname{Gamma}(\alpha_0,b),
\]
并且$\boldsymbol\Theta$与$S$独立。
\end{theorem}
\paragraph{进阶证明：变量变换与Jacobian。}
以下证明用于核验归一化常数；首次阅读可先接受定理结论，直接转到均值、方差与共轭更新。
\begin{proof}
联合密度为
\[
\prod_{i=1}^K
\frac{b^{\alpha_i}}{\Gamma(\alpha_i)}
g_i^{\alpha_i-1}e^{-bg_i}.
\]
作变量变换$g_i=s\theta_i$，$i<K$，以及
$g_K=s(1-\sum_{i<K}\theta_i)$。取值域是$s>0$与
$\boldsymbol\theta\in\Delta_{K-1}^{\circ}$，Jacobian绝对值为$s^{K-1}$。代入后联合密度分解为
\[
\underbrace{\frac{b^{\alpha_0}}{\Gamma(\alpha_0)}
s^{\alpha_0-1}e^{-bs}}_{\operatorname{Gamma}(\alpha_0,b)}
\;
\underbrace{\frac{\Gamma(\alpha_0)}
{\prod_i\Gamma(\alpha_i)}
\prod_i\theta_i^{\alpha_i-1}}_{\operatorname{Dir}(\boldsymbol\alpha)}.
\]
两个因子分别只含$s$与$\boldsymbol\theta$。第一个因子已经是归一化的
$\operatorname{Gamma}(\alpha_0,b)$密度；原始联合密度的积分为$1$，所以由
Tonelli定理，
\[
1=\int_0^\infty f_S(s)\,\mathrm ds
\int_{\Delta_{K-1}}h(\boldsymbol\theta)\,\mathrm d\boldsymbol\theta
=\int_{\Delta_{K-1}}h(\boldsymbol\theta)\,\mathrm d\boldsymbol\theta.
\]
因此第二个因子也归一化，其常数正是$1/B(\boldsymbol\alpha)$。这先证明定义
\ref{def:V5-C05-dirichlet}归一化，再由联合密度分解得到所述边缘分布与独立性，避免把待证的Dirichlet归一化当作前提。
\end{proof}

\paragraph{读前自检：均值、方差与协方差。}对$\Theta\sim\operatorname{Dir}(\alpha)$，请计算$E\Theta_i=\alpha_i/\alpha_0$与$\operatorname{Cov}(\Theta_i,\Theta_j)$，核对总和恒为1导致每行协方差和为0。\par

\begin{proposition}[均值、方差与协方差]\label{prop:V5-C05-basic-moments}
若$\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha)$，则
\begin{align}
\mathbb E(\Theta_i)
&=\frac{\alpha_i}{\alpha_0},\label{eq:V5-C05-dirichlet-mean}\\
\operatorname{Var}(\Theta_i)
&=\frac{\alpha_i(\alpha_0-\alpha_i)}
{\alpha_0^2(\alpha_0+1)},\label{eq:V5-C05-dirichlet-var}\\
\operatorname{Cov}(\Theta_i,\Theta_j)
&=-\frac{\alpha_i\alpha_j}
{\alpha_0^2(\alpha_0+1)},\quad i\ne j.
\label{eq:V5-C05-dirichlet-cov}
\end{align}
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-003 L1
\paragraph{读前自检：证明：均值、方差与协方差。}对均值、方差与协方差，条件“分别把$\Theta_i$、$\Theta_i^2$与$\Theta_i\Theta_j$乘入密度”不可省；请把$\Gamma(a+1)=a\Gamma(a)$按证明中的分解展开一层，再用各项维数闭合且需消失的交叉项确为零作检查。\par

\begin{proof}
分别把$\Theta_i$、$\Theta_i^2$与$\Theta_i\Theta_j$乘入密度，所得积分都是参数增加后的多元Beta积分。由$\Gamma(a+1)=a\Gamma(a)$得
\[
\mathbb E\Theta_i^2
=\frac{\alpha_i(\alpha_i+1)}
{\alpha_0(\alpha_0+1)},\qquad
\mathbb E(\Theta_i\Theta_j)
=\frac{\alpha_i\alpha_j}
{\alpha_0(\alpha_0+1)}.
\]
分别减去均值平方与均值乘积便得到方差和协方差。
\end{proof}

\SLAdvancedSection{一般矩、条件分布与期望对数}{sec:V5-C05-S03-advanced}
以下结果服务于变分推断和高阶性质；首次阅读可跳到第\ref{sec:V5-C05-S04}节的共轭更新。

\begin{proposition}[一般矩公式]\label{prop:V5-C05-general-moment}
若$\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha)$，且
$r_i>-\alpha_i$，则
\begin{equation}\label{eq:V5-C05-general-moment}
\mathbb E\!\left[\prod_{i=1}^K\Theta_i^{r_i}\right]
=\frac{B(\boldsymbol\alpha+\boldsymbol r)}
{B(\boldsymbol\alpha)}.
\end{equation}
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-004 L1
\paragraph{读前自检：证明：一般矩公式。}一般矩公式把问题限定在“条件$r_i>-\alpha_i$保证新积分有限”；请把$\alpha_i+r_i-1$用到的关键定义展开并完成下一步变形，检查是否得到变形没有增加前提且得到证明所述结论。\par

\begin{proof}
把乘积幂移入式\eqref{eq:V5-C05-dirichlet-density}，积分中的指数变为
$\alpha_i+r_i-1$。条件$r_i>-\alpha_i$保证新积分有限，于是
\[
\mathbb E\!\left[\prod_i\Theta_i^{r_i}\right]
=\frac1{B(\boldsymbol\alpha)}
\int_{\Delta_{K-1}}\prod_i
\theta_i^{\alpha_i+r_i-1}\,\mathrm d\boldsymbol\theta
=\frac{B(\boldsymbol\alpha+\boldsymbol r)}
{B(\boldsymbol\alpha)}.
\]
\end{proof}

\begin{proposition}[聚合与条件性质]\label{prop:V5-C05-aggregation}
把$\{1,\ldots,K\}$分成互不相交的非空组$A_1,\ldots,A_L$。若
$\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha)$，则
\[
\left(\sum_{i\in A_1}\Theta_i,\ldots,
\sum_{i\in A_L}\Theta_i\right)
\sim\operatorname{Dir}\!\left(
\sum_{i\in A_1}\alpha_i,\ldots,
\sum_{i\in A_L}\alpha_i\right).
\]
因而
\begin{equation}\label{eq:V5-C05-beta-marginal}
\Theta_i\sim\operatorname{Beta}(\alpha_i,\alpha_0-\alpha_i).
\end{equation}
此外，下式取正则条件分布版本。令
$\boldsymbol U=(\Theta_j/(1-\Theta_i))_{j\ne i}$；则对
$\operatorname{Beta}(\alpha_i,\alpha_0-\alpha_i)$几乎处处的$t\in(0,1)$（并可选取对每个$0<t<1$都成立的版本），
\begin{equation}\label{eq:V5-C05-normalized-conditional}
\mathcal L(\boldsymbol U\mid\Theta_i=t)
=\operatorname{Dir}(\boldsymbol\alpha_{-i}),
\end{equation}
且右侧不依赖$t$。等价地，该正则条件核相对于剩余单纯形测度的密度为
\[
k(t,\boldsymbol u)
=\frac{1}{B(\boldsymbol\alpha_{-i})}
  \prod_{j\ne i}u_j^{\alpha_j-1},
\qquad \boldsymbol u\in\Delta_{K-2}^{\circ},
\]
其中右端不含$t$；这避免把零概率事件$\{\Theta_i=t\}$当作普通离散条件事件。
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-005 L1
\paragraph{读前自检：证明：聚合与条件性质。}在“采用该定理中的独立Gamma表示”成立时处理聚合与条件性质：请把$\frac{\Theta_j}{1-\Theta_i}$用到的关键定义展开并完成下一步变形，并以而$\Theta_i$只由 $G_i$与该和决定，故$\boldsymbol U$与$\Theta_i$独立判定结果。\par

\begin{proof}
采用定理\ref{thm:V5-C05-gamma-normalization}中的独立Gamma表示。每一组
$H_\ell=\sum_{i\in A_\ell}G_i$是shape为$\sum_{i\in A_\ell}\alpha_i$、
rate为$b$的Gamma变量；不同组仍独立。把$H_1,\ldots,H_L$除以共同总和便得到聚合结论。取两组$\{i\}$与其补集得到式\eqref{eq:V5-C05-beta-marginal}。另一方面，
\[
\frac{\Theta_j}{1-\Theta_i}
=\frac{G_j}{\sum_{\ell\ne i}G_\ell},\qquad j\ne i.
\]
该归一化比例服从$\operatorname{Dir}(\boldsymbol\alpha_{-i})$，并独立于
$\sum_{\ell\ne i}G_\ell$；它还独立于$G_i$。而$\Theta_i$只由
$G_i$与该和决定，故$\boldsymbol U$与$\Theta_i$独立。于是可把无条件Dirichlet密度选作上述正则条件核；该核对边缘分布几乎处处唯一，并且此版本对所有$0<t<1$都有定义。
\end{proof}

\begin{proposition}[期望对数]\label{prop:V5-C05-expected-log}
若$\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha)$，则
\begin{equation}\label{eq:V5-C05-expected-log}
\mathbb E[\log\Theta_i]
=\psi(\alpha_i)-\psi(\alpha_0).
\end{equation}
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-006 L1
\paragraph{读前自检：证明：期望对数。}Dirichlet期望对数由归一化常数求导得到；请对$\log B(\alpha)$关于$\alpha_i$求导，核对$E[\log\Theta_i]=\psi(\alpha_i)-\psi(\alpha_0)$。\par

\begin{proof}
归一化恒等式为
\[
B(\boldsymbol\alpha)
=\int_{\Delta_{K-1}}\prod_j\theta_j^{\alpha_j-1}
\,\mathrm d\boldsymbol\theta.
\]
在所有$\alpha_j>0$的紧子区域内可用一个可积上界控制参数导数；对
$\alpha_i$求导并除以$B(\boldsymbol\alpha)$得到
\[
\frac{\partial}{\partial\alpha_i}\log B(\boldsymbol\alpha)
=\mathbb E[\log\Theta_i].
\]
再由式\eqref{eq:V5-C05-multivariate-beta}求导：
\[
\frac{\partial}{\partial\alpha_i}\log B(\boldsymbol\alpha)
=\psi(\alpha_i)-\psi(\alpha_0),
\]
故式\eqref{eq:V5-C05-expected-log}成立。
\end{proof}

{p(\boldsymbol n\mid\boldsymbol\alpha)}\notag\\
&\propto
\prod_i\theta_i^{n_i}
\prod_i\theta_i^{\alpha_i-1}\notag\\
&=\prod_i\theta_i^{\alpha_i+n_i-1}.
\label{eq:V5-C05-posterior-kernel}
\end{align}
令$\boldsymbol\alpha^{\mathrm{post}}=\boldsymbol\alpha+\boldsymbol n$，对式
\eqref{eq:V5-C05-posterior-kernel}在单纯形上积分得到$B(\boldsymbol\alpha^{\mathrm{post}})$。

% KNOWLEDGE-PLAN: KN-V5-C34-THEOREM-003 L1
\paragraph{读前自检：共轭。}Dirichlet--多项共轭要求$\alpha_i>0$且计数$n_i\ge0$；请合并先验与似然的幂指数，核对$\alpha+n$仍严格为正并定义合法Dirichlet后验。\par

\begin{theorem}[\DirichletMultinomial{}共轭]\label{thm:V5-C05-conjugacy}
在模型\eqref{eq:V5-C05-bayes-model}中，若
$\boldsymbol\alpha\in(0,\infty)^K$且
$\boldsymbol n\in\mathcal C_{N,K}$，则
\begin{equation}\label{eq:V5-C05-conjugate-posterior}
\boldsymbol\Theta\mid\boldsymbol n,\boldsymbol\alpha
\sim\operatorname{Dir}(\boldsymbol\alpha+\boldsymbol n).
\end{equation}
\emph{进阶旁支（边际证据）。}若还需比较模型或核验计数概率，则计数向量的边际概率为
\begin{equation}\label{eq:V5-C05-dirichlet-multinomial}
p(\boldsymbol n\mid\boldsymbol\alpha)
=\frac{N!}{\prod_i n_i!}
\frac{B(\boldsymbol\alpha+\boldsymbol n)}
{B(\boldsymbol\alpha)}.
\end{equation}
\end{theorem}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-007 L1
\paragraph{读前自检：证明：共轭。}证明共轭时只保留关于$\theta$的核；请计算$\prod_i\theta_i^{\alpha_i-1}\prod_i\theta_i^{n_i}$，核对指数为$\alpha_i+n_i-1$并由Beta常数归一化。\par

\begin{proof}
式\eqref{eq:V5-C05-posterior-kernel}与定义
\ref{def:V5-C05-dirichlet}的核相同，参数为
$\boldsymbol\alpha^{\mathrm{post}}=\boldsymbol\alpha+\boldsymbol n$。所有
$\alpha_i^{\mathrm{post}}>0$，所以归一化积分有限且为$B(\boldsymbol\alpha^{\mathrm{post}})$，由此得到
\[
p(\boldsymbol\theta\mid\boldsymbol n,\boldsymbol\alpha)
=\frac1{B(\boldsymbol\alpha+\boldsymbol n)}
\prod_i\theta_i^{\alpha_i+n_i-1}.
\]
对联合分布积分则有
\begin{align*}
p(\boldsymbol n\mid\boldsymbol\alpha)
&=\int_{\Delta_{K-1}}
p(\boldsymbol n\mid\boldsymbol\theta)
p(\boldsymbol\theta\mid\boldsymbol\alpha)
\,\mathrm d\boldsymbol\theta\\
&=\frac{N!}{\prod_i n_i!}
\frac1{B(\boldsymbol\alpha)}
\int_{\Delta_{K-1}}\prod_i
\theta_i^{\alpha_i+n_i-1}\,\mathrm d\boldsymbol\theta\\
&=\frac{N!}{\prod_i n_i!}
\frac{B(\boldsymbol\alpha+\boldsymbol n)}
{B(\boldsymbol\alpha)}.
\end{align*}
这同时给出后验归一化分母与计数向量证据。
\end{proof}

图\ref{fig:V5-C05-conjugate-update}把先验形状、观测计数与后验形状按分量对齐；相加发生在参数上，不是把两个概率向量相加。

\paragraph{读前自检：后验预测。}Dirichlet后验为$\operatorname{Dir}(\alpha+n)$；请对下一类别概率取后验期望，核对$P(Y_{N+1}=i\mid n,\alpha)=(\alpha_i+n_i)/(\alpha_0+N)$且总和为1。\par

\begin{proposition}[后验预测]\label{prop:V5-C05-predictive}
在定理\ref{thm:V5-C05-conjugacy}的条件下，新观测的后验预测概率为
\begin{equation}\label{eq:V5-C05-one-step-predictive}
\Pr(Y_{N+1}=i\mid\boldsymbol n,\boldsymbol\alpha)
=\frac{\alpha_i+n_i}{\alpha_0+N}.
\end{equation}
若未来$M$次观测的计数为$\boldsymbol r\in\mathcal C_{M,K}$，则
\begin{equation}\label{eq:V5-C05-future-count-predictive}
\Pr(\boldsymbol r\mid\boldsymbol n,\boldsymbol\alpha)
=\frac{M!}{\prod_i r_i!}
\frac{B(\boldsymbol\alpha+\boldsymbol n+\boldsymbol r)}
{B(\boldsymbol\alpha+\boldsymbol n)}.
\end{equation}
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C34-PROOF-008 L1
\paragraph{读前自检：证明：后验预测。}后验预测证明用全概率公式积分$P(Y_{N+1}=i\mid\theta)=\theta_i$；请对$\operatorname{Dir}(\alpha+n)$求$E\Theta_i$，核对得到$(\alpha_i+n_i)/(\alpha_0+N)$。\par

\begin{proof}
由全概率公式和式\eqref{eq:V5-C05-dirichlet-mean}，
\[
\Pr(Y_{N+1}=i\mid\boldsymbol n,\boldsymbol\alpha)
=\int\theta_i\,
p(\boldsymbol\theta\mid\boldsymbol n,\boldsymbol\alpha)
\,\mathrm d\boldsymbol\theta
=\frac{\alpha_i+n_i}{\alpha_0+N}.
\]
把后验$\operatorname{Dir}(\boldsymbol\alpha+\boldsymbol n)$视作未来计数的先验，再应用式\eqref{eq:V5-C05-dirichlet-multinomial}，得到
\eqref{eq:V5-C05-future-count-predictive}。
\end{proof}

\textbf{概率解释。}把$\alpha_i$看作更新规则中的先验质量，则观测类别$i$一次就把该分量加一。式\eqref{eq:V5-C05-one-step-predictive}等价于Pólya瓮规则：当前类别$i$的质量为$\alpha_i+n_i$，总质量为$\alpha_0+N$；抽到一个类别后把对应质量再加一。因此未来观测虽然在给定
$\boldsymbol\Theta$后条件独立，积分掉$\boldsymbol\Theta$以后却具有正的同类强化，并且预测序列仍可交换。

\phantomsection\label{struct:V5-C05-CH16}
\textbf{优化解释与边界。}狄利克雷后验均值是
\[
\widehat\theta_i^{\mathrm{mean}}
=\frac{\alpha_i+n_i}{\alpha_0+N}.
\]
若所有$\alpha_i^{\mathrm{post}}=\alpha_i+n_i>1$，拉格朗日乘子法给出唯一内点众数
\begin{equation}\label{eq:V5-C05-map}
\widehat\theta_i^{\mathrm{MAP}}
=\frac{\alpha_i^{\mathrm{post}}-1}{\alpha_0^{\mathrm{post}}-K}.
\end{equation}
若任一$\alpha_i^{\mathrm{post}}\le1$，式\eqref{eq:V5-C05-map}的内点条件失效；峰值可能位于边界，某些方向的密度还会无界。“$\alpha_i$是伪计数”适合描述后验更新与预测；“$\alpha_i-1$是众数修正”只在内点众数条件成立时使用，二者不能互换。

\begin{solution}
\SLReadTranslation{题目要求完成“三分类先验的矩、更新与预测”：依据题面概率模型求出目标量，并解释条件化、归一化或边缘化。}
\SolGiven 题面概率、权重或密度均处于合法范围；条件分母/归一化常数应为正，支持集与随机变量取值域保持一致。
\SLMethodTrigger{先写支持集、条件分母或归一化常数，再代入概率公式并做概率和/边缘核验。}
\SolDerive
\textbf{计算。}先验与后验更新链为
\[
\alpha_0=2+3+5=10,\qquad
\boldsymbol\alpha^{\mathrm{post}}
=\boldsymbol\alpha+\boldsymbol n=(6,4,5),\qquad
\alpha_0^{\mathrm{post}}=15.
\]
因此
\[
\mathbb E(\boldsymbol\Theta)
=\left(\frac15,\frac3{10},\frac12\right),
\qquad
\mathbb E(\boldsymbol\Theta\mid\boldsymbol n)
=\left(\frac25,\frac4{15},\frac13\right).
\]
后一向量也是下一次观测的预测概率。所有$\alpha_i^{\mathrm{post}}>1$，所以后验MAP为
\[
\left(\frac5{12},\frac14,\frac13\right).
\]
未来依次出现类别$1$、类别$2$的概率是
\[
\frac6{15}\frac4{16}=\frac1{10};
\]
若只关心未来两次中两个类别各出现一次，则还要计入两种顺序，概率为$1/5$。

\textbf{独立核验。}后验预测分量和为$2/5+4/15+1/3=1$；反向顺序的概率为$(4/15)(6/16)=1/10$，与正向顺序相加确为$1/5$。

\textbf{结论。}后验为$\operatorname{Dir}(6,4,5)$，下一次预测为$(2/5,4/15,1/3)$，未来类别1、2各一次的无序概率为$1/5$。
\end{solution}


\phantomsection\label{struct:V5-C05-CH19}
\textbf{伪代码。}算法\ref{alg:V5-C05-closed-form-update}是由共轭公式派生的确定性闭式流程，不是需要迭代至收敛的优化程序。它接受原始类别序列或经验证的计数向量，二者只能选择一种；算法保留完整组合系数，因此同一过程既能给后验参数，也能给计数证据与未来计数预测。

\phantomsection\label{struct:V5-C05-CH20}
\textbf{输入与输出。}输入是类别数$K$、严格正先验参数
$\boldsymbol\alpha$、原始序列$\boldsymbol y$或计数向量
$\boldsymbol n$，以及可选的未来计数$\boldsymbol r$。输出为经核验的
$N$与$\boldsymbol n$、后验参数$\boldsymbol\alpha^{\mathrm{post}}$、完整预测向量、可选计数向量对数证据和可选未来计数对数概率。

\phantomsection\label{struct:V5-C05-CH21}
\textbf{参数含义。}$K$决定向量维数；$\alpha_i>0$保证先验可正规化；
$\alpha_0$与$\alpha_0^{\mathrm{post}}$分别是先验和后验总浓度；$N$是已观测样本数；
$M=\sum_i r_i$是未来计数总数。数值实现采用
$\log\Gamma$累加计算$\log B$，不直接形成可能上溢的阶乘与Gamma乘积。

\phantomsection\label{struct:V5-C05-CH22}
\textbf{初始化。}先核验$K\ge1$、$\boldsymbol\alpha$长度为$K$且所有分量有限严格为正。若输入序列，初始化
$n_i\leftarrow0$；若输入计数，复制为整数工作向量并核验非负。空序列或全零计数是合法的$N=0$情形，后验等于先验；$K=1$是退化零维单纯形，唯一概率为$1$。
这里的$K=1$只是算法对退化情形的显式扩展，不属于定义
\ref{def:V5-C05-dirichlet}所规定的$K\ge2$非退化Dirichlet分布。

\phantomsection\label{struct:V5-C05-CH23}
\textbf{循环变量与更新顺序。}序列输入时，循环变量$t=1,\ldots,N$，按观测顺序核验
$y_t\in\{1,\ldots,K\}$并执行$n_{y_t}\leftarrow n_{y_t}+1$。计数完成后，按
$i=1,\ldots,K$依次更新$\alpha_i^{\mathrm{post}}\leftarrow\alpha_i+n_i$，再统一计算
$\alpha_0^{\mathrm{post}}$与预测概率$\alpha_i^{\mathrm{post}}/\alpha_0^{\mathrm{post}}$。未来计数只用于预测，不得混入当前后验。

\phantomsection\label{struct:V5-C05-CH24}
\textbf{判断条件与停止条件。}类别序列与计数向量必须恰好提供一种：仅序列时核验类别并累计计数，仅计数时核验后复制；两者都给或都不给均返回$\mathtt{invalid\_input}$。任何维数不符、非有限超参数、非正
$\alpha_i$、非法类别、非整数或负计数、总数不一致、未来计数非法，也都以$\mathtt{invalid\_input}$停止并附具体诊断；停止前的部分计数不作为后验输出。两条合法分支汇合后才统一计算$N$、后验参数、证据与预测。正常情形在一次输入扫描和一次长度$K$的向量扫描后结束，没有容差型终止条件。

\phantomsection\label{struct:V5-C05-CH25}
\textbf{正确性依据与收敛条件。}正确性依据是定理
\ref{thm:V5-C05-conjugacy}和命题\ref{prop:V5-C05-predictive}。计数向量对数证据由式
\eqref{eq:V5-C05-dirichlet-multinomial}在对数域稳定求值。该闭式算法没有渐近迭代，因此“迭代收敛条件”不适用；完成有限次精确算术操作即得到理论值。浮点输出仍需满足有限性、概率非负以及预测和在舍入容差内等于$1$。不收敛或不适用情形是：把它误用于非多项似然、非Dirichlet先验、含负权计数的数据，或要求从未知复杂后验产生样本。

\phantomsection\label{struct:V5-C05-CH26}
\textbf{训练时间复杂度。}从原始序列计数并形成后验的训练时间复杂度为
$O(N+K)$；若已经提供合法计数，则为$O(K)$。计算计数边际证据再做常数次
$K$项$\log\Gamma$求和，仍为$O(K)$。

\phantomsection\label{struct:V5-C05-CH27}
\textbf{预测时间复杂度与空间复杂度。}形成全部$K$类预测概率的预测时间复杂度为
$O(K)$；在已保存$\alpha_i^{\mathrm{post}},\alpha_0^{\mathrm{post}}$后查询单个类别是$O(1)$。计算一个长度
$K$的未来计数概率为$O(K)$。工作空间复杂度为$O(K)$；若允许原地覆盖调用方计数可减少常数，但不能破坏原始计数输入。

核验$K$是正整数、$\boldsymbol\alpha$长度为$K$且分量有限严格为正；失败则返回$\mathtt{invalid\_input}$及字段诊断\;
\uIf{仅提供类别序列$\boldsymbol y$}{
  初始化$n_i\leftarrow0$，$i=1,\ldots,K$\;
  \For{$t\leftarrow1,\ldots,|\boldsymbol y|$}{
    $w\leftarrow w+1$\;
    若$y_t\notin\{1,\ldots,K\}$，则返回$\mathtt{invalid\_input}$、诊断$\mathtt{invalid\_class}(t)$且不返回部分后验\;
    $n_{y_t}\leftarrow n_{y_t}+1$\;
  }
}
\uElseIf{仅提供计数向量$\boldsymbol n$}{
  $w\leftarrow w+K$\;
  若其长度不是$K$，则返回最后合法空工作状态、$\mathtt{invalid\_input}$及$\mathtt{length\_mismatch}$诊断\;
  若任一分量不是非负整数，则返回最后合法空工作状态、$\mathtt{invalid\_input}$及$\mathtt{invalid\_count}(i)$诊断；否则复制为工作向量\;
}
\Else{
  返回$\mathtt{invalid\_input}$，并诊断“类别序列与计数向量必须恰好提供一种”\;
}
令$N\leftarrow\sum_i n_i$；此后两种输入共用同一后验、证据与预测分支\;
若提供$N_{\rm in}$但$N_{\rm in}\ne N$，则返回$\mathtt{invalid\_input}$及$\mathtt{total\_mismatch}$诊断\;
\For{$i\leftarrow1,\ldots,K$}{
  $w\leftarrow w+1$\;
  在临时向量中令$\alpha_i^+\leftarrow\alpha_i+n_i$\;
}
令$\alpha_0^+=\sum_i\alpha_i^+$，$p_i^+=\alpha_i^+/\alpha_0^+$\;
用$\log\Gamma$计算临时计数对数证据
$\log N!-\sum_i\log n_i!+\log B(\boldsymbol\alpha^+)-\log B(\boldsymbol\alpha)$\;
若提供$\boldsymbol r$，逐项核验并增加$w$；非法时返回计数工作状态、$\mathtt{invalid\_input}$及未来计数位置诊断；否则在对数域形成临时未来概率\;
判断全部临时输出有限、预测非负且和在容差内为$1$；失败则返回计数工作状态、$\mathtt{numerical\_failure}$及特殊函数/预测诊断\;
原子提交$(\boldsymbol\alpha^{\mathrm{post}},\alpha_0^{\mathrm{post}},\boldsymbol p_{\mathrm{pred}})$及可选对数概率；返回完整结果、$w,\mathtt{completed}$及归一化/退化最终诊断\;
\end{algorithm}
\end{AlgorithmContract}
\FloatBarrier
\SLRobustImplementationStrip{核心流程跟踪先验参数加计数、得到后验、读取均值与后验预测；特殊函数、空计数和边界状态属于稳健实现细节}
\endgroup

\begin{notapplicablebox}[title=\SLComplexityAssumptionsTitle]
类别索引访问、浮点加法和一次$\log\Gamma$求值按固定精度视为
$O(1)$；输入是稠密长度$K$向量。若使用任意精度算术，单次特殊函数成本随目标精度增长，不能再把上述界理解为位复杂度。
\end{notapplicablebox}

\phantomsection\label{struct:V5-C05-CH28}
\textbf{适用条件。}该方法适合有限互斥类别、固定概率向量、条件独立类别观测和
Dirichlet先验。它也适合作为更大概率模型中的局部共轭更新。若概率随样本协变量变化、类别并非互斥、观测带非整数权重或先验不是Dirichlet，必须重新推导后验，不能机械相加。

\phantomsection\label{struct:V5-C05-CH29}
\textbf{局限性。}共轭性提高可计算性，却不自动表达类别间的任意相关结构；单个
Dirichlet只能给出由和约束产生的负协方差。固定$\boldsymbol\alpha$也可能过强地平滑稀有类别。层次Dirichlet、混合分布或协变量模型可以扩大表达能力，但相应超参数推断通常不再保持本章全部闭式性质。

\phantomsection\label{struct:V5-C05-CH30}
\textbf{常见错误。}不得把$B(\boldsymbol\alpha)$与其倒数互换；不得在计数证据中丢掉多项组合系数；不得允许$\alpha_i=0$后仍声称先验可正规化；不得把
$\alpha_0$和均值向量混成同一作用；不得写
$\mathbb E\log\Theta_i=\log\mathbb E\Theta_i$；不得在$\alpha_i^{\mathrm{post}}\le1$时套用内点
MAP公式；多步预测必须在每一步更新分子和分母。

\begin{exercise}\label{ex:V5-C05-S03-02}

$(G_1,\ldots,G_K)\mapsto(S,\Theta_1,\ldots,\Theta_{K-1})$的取值域和Jacobian，并证明$S$与$\boldsymbol\Theta$独立。若各分量rate不同，指出原证明在哪一步不再分离。
\end{exercise}
\ifSLFullEdition
\phantomsection\label{sol:V5-C05-S03-02}
\begin{chapterexercisesolution}{ex:V5-C05-S03-02}
\nopagebreak[4]
设$K\ge2$、$\alpha_i>0$、$b>0$，且$G_1,\ldots,G_K$相互独立并满足$G_i\sim\operatorname{Gamma}(\alpha_i,b)$。记$\alpha_0=\sum_i\alpha_i$，其完整联合密度为
\[
f_{\boldsymbol G}(\boldsymbol g)
=\frac{b^{\alpha_0}}{\prod_{i=1}^K\Gamma(\alpha_i)}
 \left(\prod_{i=1}^K g_i^{\alpha_i-1}\right)
 e^{-b\sum_i g_i}\boldsymbol1_{(0,\infty)^K}(\boldsymbol g).
\]
令$S=\sum_iG_i$、$\Theta_i=G_i/S$。从$(s,\theta_1,\ldots,\theta_{K-1})$到$(g_1,\ldots,g_K)$的变换为$g_i=s\theta_i$，其中
$s>0$、$\theta_i>0$且$\sum_{i<K}\theta_i<1$，并置
$\theta_K=1-\sum_{i<K}\theta_i$；Jacobian绝对值为$s^{K-1}$。共同rate $b$使指数项变为
\[
\exp\!\left(-b\sum_i s\theta_i\right)=e^{-bs},
\]
而幂次与Jacobian合并为$s^{\alpha_0-1}\prod_i\theta_i^{\alpha_i-1}$。所以变换后的联合密度逐因子分解为
\[
\underbrace{\frac{b^{\alpha_0}}{\Gamma(\alpha_0)}
s^{\alpha_0-1}e^{-bs}}_{\operatorname{Gamma}(\alpha_0,b)\text{密度}}
\;
\underbrace{\frac{\Gamma(\alpha_0)}{\prod_i\Gamma(\alpha_i)}
\prod_{i=1}^K\theta_i^{\alpha_i-1}}_{\operatorname{Dir}(\boldsymbol\alpha)\text{密度}}.
\]
两因子分别只依赖$s$与$\boldsymbol\theta$，且各自积分为$1$，故$S\sim\operatorname{Gamma}(\alpha_0,b)$、$\boldsymbol\Theta\sim\operatorname{Dir}(\boldsymbol\alpha)$，并且二者独立。

若rate为$b_i$，指数项成为
\[
\exp\!\left(-s\sum_i b_i\theta_i\right),
\]
同时含$s$和$\boldsymbol\theta$，不再能按上述方式分离。归一化比例一般也不再服从参数仅为$\boldsymbol\alpha$的Dirichlet分布。
\end{chapterexercisesolution}

\begin{exercise}\label{ex:V5-C05-S05-01}
先验参数为$(2,3,5)$，当前计数为$(4,1,0)$。求未来依次出现类别$1$、类别$2$的概率，以及未来两次中两个类别各出现一次而不计顺序的概率。说明为何不能把当前单步预测的两个分量直接相乘。
\end{exercise}
\ifSLFullEdition
\phantomsection\label{sol:V5-C05-S05-01}
\begin{chapterexercisesolution}{ex:V5-C05-S05-01}
\nopagebreak[4]
当前后验参数是$(6,4,5)$，总和为$15$。指定顺序$1$后$2$的概率为
\[
\frac6{15}\frac4{16}=\frac1{10}.
\]
反序$2$后$1$的概率是
$\frac4{15}\frac6{16}=1/10$，所以
\[
P(\{1,2\}\text{各一次})
=\frac6{15}\frac4{16}
+\frac4{15}\frac6{16}
=\frac15.
\]
不能直接把当前预测分量
$6/15$与$4/15$相乘，因为第一次未来观测后总质量由$15$变为$16$，对应类别的分子也可能更新。
\end{chapterexercisesolution}
\fi

\begin{exercise}\label{ex:V5-C05-S05-02}
后验参数为$\boldsymbol\alpha^{\mathrm{post}}=(1/2,3,3/2)$。求后验均值；判断式

$\operatorname{Dir}(2,3,5)$与$\operatorname{Dir}(20,30,50)$的均值和方差，说明“伪计数”“众数修正”“浓度”三种解释的边界。
\end{exercise}
\ifSLFullEdition
\phantomsection\label{sol:V5-C05-S05-02}
\begin{chapterexercisesolution}{ex:V5-C05-S05-02}
\nopagebreak[4]
$\alpha_0^{\mathrm{post}}=5$，后验均值为
\[
\left(\frac1{10},\frac35,\frac3{10}\right).
\]
因为$\alpha_1^{\mathrm{post}}=1/2\le1$，内点条件不成立，式
\eqref{eq:V5-C05-map}不能使用。密度含因子
$\theta_1^{-1/2}$，在保持其他分量为正并令$\theta_1\downarrow0$时趋于无穷；峰值位于边界意义下。

$\operatorname{Dir}(2,3,5)$与
$\operatorname{Dir}(20,30,50)$均值同为
$(1/5,3/10,1/2)$，但后者总浓度从$10$增至$100$，方差按
$1/(\alpha_0+1)$型因子显著缩小。更新和预测中的先验质量是
$\alpha_i$；内点众数的指数修正是$\alpha_i-1$；总浓度是
$\alpha_0$。这三种解释作用于不同公式。
\[
\operatorname{Var}(\Theta_i)
=\frac{\mu_i(1-\mu_i)}{\alpha_0+1},\qquad
\frac{\operatorname{Var}_{(20,30,50)}(\Theta_i)}
{\operatorname{Var}_{(2,3,5)}(\Theta_i)}
=\frac{11}{101}.
\]
\end{chapterexercisesolution}
