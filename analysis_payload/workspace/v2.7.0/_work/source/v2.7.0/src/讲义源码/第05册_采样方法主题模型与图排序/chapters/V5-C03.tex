\phantomsection\label{struct:V5-C03-CH03}
\begin{precheckbox}
下列各项中，$K$为马尔可夫核，$\pi$为候选平稳分布，$I_2$为二阶单位矩阵，$a,b\ge0$。
\begin{enumerate}[leftmargin=2.2em]
  \item 给定核$K$与分布$\pi$，能否逐项核验$\pi K=\pi$，并说明这只意味着平稳？
  \item 能否为恒等转移矩阵$I_2$写出全部平稳分布，并据此区分可逆、不可约与唯一平稳分布？
  \item 能否把$E_\pi f$写成样本平均，并说明独立样本方差公式不能直接用于相关轨迹？
  \item 能否证明$\min\{a,b\}$的交换对称性，并判断非负函数何时能归一化为概率密度？
\end{enumerate}
第1--2项不足时复习\GlobalChapterRef{30}{5}{1}，第3项不足时复习\GlobalChapterRef{31}{5}{2}；第4项直接支撑MH流量证明与目标合法性检查。
\end{precheckbox}

\phantomsection\label{struct:V5-C03-CH06}
设目标是计算
\begin{equation}\label{eq:V5-C03-target-expectation}
I=E_\pi[f(X)]=\int_{\mathcal X}f(x)\pi(x)\,\mu(\mathrm dx),
\end{equation}
但不能直接产生独立的$\pi$样本。MCMC不把相关状态误称为独立样本，而是设计一个转移核$K$，使$\pi$保持不变，再以同一条链的时间平均
\begin{equation}\label{eq:V5-C03-ergodic-average}
\widehat I_{m,n}
=\frac1n\sum_{t=m+1}^{m+n}f(X_t),\qquad n\ge1,
\end{equation}
估计式\eqref{eq:V5-C03-target-expectation}。式中$m$不是证明链已经收敛的参数；丢弃前$m$步只能减弱某些初值影响，不能修复可约、周期、错误目标或错误转移核。

\phantomsection\label{struct:V5-C03-CH07}
\begin{definition}[MCMC构造]
给定目标概率分布$\pi$和函数$f$，若马尔可夫核$K$满足$\pi K=\pi$，并且链的结构条件足以保证式\eqref{eq:V5-C03-ergodic-average}收敛到$E_\pi f$，则“按$K$产生轨迹并计算时间平均”的方法称为以$\pi$为目标的马尔可夫链蒙特卡罗法。
\end{definition}
定义包含两个层次：$\pi K=\pi$是目标保持条件；从给定初值出发的可达性、回返与遍历条件负责把时间平均连接到目标期望。只验证第一层不足以宣称估计一致。

\phantomsection\label{struct:V5-C03-CH08}
\textbf{模型组成与合法性。}本章的数学对象是五元组
\[
\bigl((\mathcal X,\mathcal B),\mu,\widetilde\pi,q,\nu_0\bigr).
\]
要求$(\mathcal X,\mathcal B)$是标准Borel空间，$\mu$是$\sigma$有限测度；$\widetilde\pi$可测且$0<Z=\int\widetilde\pi\,\mathrm d\mu<\infty$，从而$\pi=\widetilde\pi/Z$是概率密度；$q:\mathcal X\times\mathcal X\to[0,\infty)$关于$\mathcal B\otimes\mathcal B$联合可测，且对每个合法$x$都有$\int q(x,y)\mu(\mathrm dy)=1$；初始分布$\nu_0$落在$S=\{x:0<\widetilde\pi(x)<\infty\}$内。联合可测性保证$x\mapsto\int_Bq(x,y)\mu(\mathrm dy)$对每个$B\in\mathcal B$可测，故$q$确实诱导提议马尔可夫核。若$\widetilde\pi$不可归一化，未知常数$Z$无法成为“会在比值中消去”的许可证，因为此时目标概率分布根本不存在。

\begin{warningbox}
\textbf{先保留数学主线。}细致平衡只推出平稳，不推出不可约、唯一平稳或遍历；另外，MH目标核必须存在有限正归一化常数。两个最小反例分别是恒等核$K=I_2$和不可积的非负目标核。来源页码与核验过程统一移到章末“核验记录”，不插入算法推导。
\end{warningbox}

\textbf{学习策略。}MCMC在统计学习中的策略不是直接最小化一个经验风险，而是把后验期望、预测积分或隐变量期望写成式\eqref{eq:V5-C03-target-expectation}，再同时满足四项设计目标：
\begin{enumerate}[leftmargin=2.2em]
  \item 目标保持：$\pi K=\pi$；
  \item 可达性：目标支持内的重要区域属于同一可通信部分；
  \item 可计算性：只使用$\widetilde\pi$的点值、提议抽样和提议密度比；
  \item 统计效率：在正确性不受损的前提下降低相关性与计算成本。
\end{enumerate}
第1项是代数条件，第2项是图或测度结构条件，第3项是可计算条件，第4项才涉及调参。高接受率、漂亮轨迹或小的经验MCSE都不能替代前3项。

\phantomsection\label{struct:V5-C03-CH11}
% KNOWLEDGE-PLAN: KN-V5-C32-DEFINITION-002 L1
\paragraph{读前自检：细致平衡。}设$\pi$为概率密度，马尔可夫核$K$有密度$k$，且$\pi(x)k(x,y)=\pi(y)k(y,x)$对$\mu\otimes\mu$几乎处处成立。请对$x$积分，核对总流入为$\pi(y)$，故$\pi K=\pi$。\par

\begin{definition}[细致平衡]
若对任意可测集合$A,B\in\mathcal B$都有
\begin{equation}\label{eq:V5-C03-general-detailed-balance}
\int_A\pi(\mathrm dx)K(x,B)
=\int_B\pi(\mathrm dy)K(y,A),
\end{equation}
则称$K$关于$\pi$满足细致平衡，或称平稳链关于$\pi$可逆。若$K$有密度$k(x,y)$，条件化为
\begin{equation}\label{eq:V5-C03-density-detailed-balance}
\pi(x)k(x,y)=\pi(y)k(y,x)
\quad\text{对}\ \mu\otimes\mu\ \text{几乎处处成立}.
\end{equation}
\end{definition}
\index{细致平衡}
\index{可逆马尔可夫链}

从局部流到平稳分布的推导分三步。第一步，式\eqref{eq:V5-C03-density-detailed-balance}把$x\to y$的平稳概率流与$y\to x$的反向流配对。第二步，对$x$积分，
\[
\int\pi(x)k(x,y)\,\mu(\mathrm dx)
=\int\pi(y)k(y,x)\,\mu(\mathrm dx).
\]
第三步使用$\int k(y,x)\mu(\mathrm dx)=1$，右端成为$\pi(y)$，于是总流入等于目标质量。图\ref{fig:V5-C03-mh-balance-flux}把双向提议流与截平后的成对接受流分开绘制。

\begin{proposition}[细致平衡推出平稳]\label{prop:V5-C03-db-stationary}
若概率测度$\pi$与马尔可夫核$K$满足式\eqref{eq:V5-C03-general-detailed-balance}，则$\pi K=\pi$。
\end{proposition}

\paragraph{读前自检：证明：细致平衡推出平稳。}细致平衡推出平稳以“在该等式中取$A=\mathcal X$”为约束；请对$(\pi K)(B)$做一次条件化或边缘化，并确认证明没有使用不可约性或非周期性，所以结论也不能附带唯一平稳分布、边缘收敛或时间平均收敛。\par

\begin{proof}
在式\eqref{eq:V5-C03-general-detailed-balance}中取$A=\mathcal X$。因为$K(y,\mathcal X)=1$，
\[
(\pi K)(B)
=\int_{\mathcal X}\pi(\mathrm dx)K(x,B)
=\int_B\pi(\mathrm dy)K(y,\mathcal X)
=\pi(B)
\]
对任意$B\in\mathcal B$成立。因此$\pi K=\pi$。证明没有使用不可约性或非周期性，所以结论也不能附带唯一平稳分布、边缘收敛或时间平均收敛。
\end{proof}

\SLTeachSection{\MetropolisHastings{}算法}{sec:V5-C03-S03}
\knowledgeanchor{SLM-19-04-003}{Metropolis-Hastings接受率@\MetropolisHastings{} 接受率}
\knowledgeanchor{SLM-19-04-004}{双向目标—提议流}
\knowledgeanchor{SLM-19-04-005}{Metropolis-Hastings转移核@\MetropolisHastings{} 转移核}
\knowledgeanchor{SLM-19-06-002}{Metropolis-Hastings核可逆性@\MetropolisHastings{} 核可逆性}
\knowledgeanchor{SLM-19-02-008}{Metropolis-Hastings拒绝自环@\MetropolisHastings{} 拒绝自环}
\knowledgeanchor{SLM-19-04-006}{单分量Metropolis-Hastings算法@单分量\MetropolisHastings{} 算法}

定义从状态$x$提议$y$的正向目标--提议流为
\begin{equation}\label{eq:V5-C03-directed-flow}
g(x,y)=\widetilde\pi(x)q(x,y).
\end{equation}
为避免在$g(x,y)=g(y,x)=0$时产生$0/0$，先定义接受后的无向公共流
\begin{equation}\label{eq:V5-C03-common-flow}
h(x,y)=\min\{g(x,y),g(y,x)\}.
\end{equation}
再在实际有正向提议流时定义
\begin{equation}\label{eq:V5-C03-mh-alpha}
\alpha(x,y)=
\begin{cases}
h(x,y)/g(x,y)
=\min\!\left\{1,
\dfrac{\widetilde\pi(y)q(y,x)}
{\widetilde\pi(x)q(x,y)}\right\},&g(x,y)>0,\\[1.1em]
0,&g(x,y)=0.
\end{cases}
\end{equation}
第二个分支的取值不会改变从$x$实际产生的提议分布；算法仍需核验抽样器返回的候选与$q$求值器一致。若$\widetilde\pi(y)=0$或$q(y,x)=0$，则$h(x,y)=0$并拒绝，不把零分子改成一个很小的正数。

由于$\widetilde\pi$与$q$联合可测，$g,h,\alpha$均可测；因$0\le q\alpha\le q$，下式中的积分是$[0,1]$值可测函数。拒绝概率和完整MH核分别为
\begin{equation}\label{eq:V5-C03-rejection-mass}
r(x)=1-\int q(x,y)\alpha(x,y)\,\mu(\mathrm dy),
\end{equation}
\begin{equation}\label{eq:V5-C03-mh-kernel}
K(x,\mathrm dy)
=q(x,y)\alpha(x,y)\mu(\mathrm dy)
+r(x)\delta_x(\mathrm dy).
\end{equation}
第一项已经包括离散情形下被接受的自提议；第二项专门记录所有拒绝造成的自环。删除重复状态会改变核和时间尺度。图\ref{fig:V5-C03-mh-accept-reject}展示提议、接受、拒绝与留驻的关系。

\noindent\textbf{读图顺序：}先看从当前状态 $X_t=x$ 按 $q(x,\cdot)$ 提出候选 $Y=y$；再看在 $g(x,y)>0$ 时计算 $\alpha(x,y)$，抽取 $U\sim\mathrm U(0,1)$ 并比较 $U\le\alpha(x,y)$；最终接受时令 $X_{t+1}=y$，拒绝时仍记录一次旧状态 $X_{t+1}=x$，形成自环。

\begin{theorem}[MH核的可逆性]\label{thm:V5-C03-mh-reversible}
若$(\mathcal X,\mathcal B)$为标准Borel空间，$\mu$为$\sigma$有限测度，$0<Z=\int\widetilde\pi\,\mathrm d\mu<\infty$，且$q$是上述联合可测的合法提议密度，则式\eqref{eq:V5-C03-mh-alpha}--\eqref{eq:V5-C03-mh-kernel}定义一个马尔可夫核$K$。该核关于$\pi=\widetilde\pi/Z$满足细致平衡，因而$\pi$是$K$的平稳分布。
\end{theorem}
% KNOWLEDGE-PLAN: KN-V5-C32-PROOF-002 L1
\paragraph{读前自检：证明：MH核的可逆性。}MH接受流满足$\widetilde\pi(x)q(x,y)\alpha(x,y)=h(x,y)=h(y,x)$；请除以共同归一化常数并对$A\times B$积分，核对接受移动部分双向流相等，再验证拒绝自环项也满足详细平衡。\par

\begin{proof}
由式\eqref{eq:V5-C03-common-flow}--\eqref{eq:V5-C03-mh-alpha}，对$\mu\otimes\mu$几乎处处的$(x,y)$，
\[
\widetilde\pi(x)q(x,y)\alpha(x,y)
=h(x,y)
=h(y,x)
=\widetilde\pi(y)q(y,x)\alpha(y,x).
\]
除以共同的正归一化常数$Z$，接受移动部分的双向流相等。对任意$A,B\in\mathcal B$，接受部分满足
\[
\int_A\!\int_B\pi(x)q(x,y)\alpha(x,y)
\,\mu(\mathrm dy)\mu(\mathrm dx)
=\int_B\!\int_A\pi(y)q(y,x)\alpha(y,x)
\,\mu(\mathrm dx)\mu(\mathrm dy).
\]
拒绝自环部分在两侧都等于
\[
\int_{A\cap B}\pi(x)r(x)\,\mu(\mathrm dx),
\]
所以完整核满足式\eqref{eq:V5-C03-general-detailed-balance}。命题\ref{prop:V5-C03-db-stationary}再给出$\pi K=\pi$。此证明只建立可逆与平稳，没有建立不可约、唯一或遍历。
\end{proof}

\paragraph{读前自检：接受率的成对流推导。}成对接受流令$a=g(x,y),b=g(y,x)$且$0\le\alpha_{xy},\alpha_{yx}\le1$；请由$c=a\alpha_{xy}=b\alpha_{yx}\le\min\{a,b\}$构造取等的双向接受率，并核对$a=0$或$b=0$时不计算$0/0$。\par

\SLTeachSection{接受率推导}{sec:V5-C03-S04}
\knowledgeanchor{PRE-SA-007}{接受率的成对流推导}
\begin{derivationgoalbox}
从细致平衡的成对流约束出发，推导在固定提议核下逐对最大的合法接受流，并由此得到包含零流边界的\MetropolisHastings{}接受概率。
\end{derivationgoalbox}
\begin{derivationprepbox}
固定$x,y$，记正向与反向提议流为$a=g(x,y)$和$b=g(y,x)$。接受概率必须属于$[0,1]$；若要求已接受流满足细致平衡，则$a\alpha_{xy}=b\alpha_{yx}$。
\end{derivationprepbox}
\textbf{推导第1步：给出共同流的上界。}
令$c=a\alpha_{xy}=b\alpha_{yx}$。由$0\leq\alpha_{xy},\alpha_{yx}\leq1$，同时有$c\leq a$与$c\leq b$，因而$c\leq\min\{a,b\}$。

\textbf{推导第2步：构造达到上界的双向接受概率。}
当$a,b>0$时取$\alpha_{xy}=\min\{a,b\}/a$及$\alpha_{yx}=\min\{a,b\}/b$；若某一方向流为零，则共同流只能为零，并把由正流指向零反向流的移动确定拒绝。该构造在所有边界情形均不需要计算$0/0$。

\textbf{推导第3步：还原MH接受核。}
在$a>0$的分支，正向接受概率化为$\alpha(x,y)=\min\{1,b/a\}$；代回$a=\widetilde\pi(x)q(x,y)$与$b=\widetilde\pi(y)q(y,x)$即得式\eqref{eq:V5-C03-mh-alpha}。未接受的剩余质量必须并入$x$处拒绝自环，才能得到总质量为$1$的马尔可夫核。

% KNOWLEDGE-PLAN: KN-V5-C32-PROPOSITION-002 L1
\paragraph{读前自检：成对约束下的最大接受流。}成对约束下的最大接受流要求先确认“$0\le\alpha_{xy},\alpha_{yx}\le1, \qquad a\alpha_{xy}=b\alpha_{yx}$”；请随后把$0\le\alpha_{xy},\alpha_{yx}\le1, \qquad a\alpha_{xy}$移到同一边并核对符号，用该等式达到该上界验收。\par

\begin{proposition}[成对约束下的最大接受流]\label{prop:V5-C03-maximal-flow}
固定一对状态$x,y$并记$a=g(x,y)$、$b=g(y,x)$。在所有满足
\[
0\le\alpha_{xy},\alpha_{yx}\le1,
\qquad a\alpha_{xy}=b\alpha_{yx}
\]
的接受规则中，共同流$a\alpha_{xy}=b\alpha_{yx}$的最大值为$\min\{a,b\}$；式\eqref{eq:V5-C03-mh-alpha}达到该上界。
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C32-PROOF-003 L1
\paragraph{读前自检：证明：成对约束下的最大接受流。}从成对约束下的最大接受流的条件“若$a>0$，取$\alpha_{xy}=\min\{a,b\}/a$”出发，请把$\alpha_{xy}=\min\{a,b\}/a$用到的关键定义展开并完成下一步变形；所得量应符合变形没有增加前提且得到证明所述结论。\par

\begin{proof}
设共同流为$c$。由$\alpha_{xy}\le1$得$c=a\alpha_{xy}\le a$；由$\alpha_{yx}\le1$得$c=b\alpha_{yx}\le b$，故$c\le\min\{a,b\}$。若$a>0$，取$\alpha_{xy}=\min\{a,b\}/a$；若$a=0$，取$\alpha_{xy}=0$。对反向作同样定义，便有两侧共同流都等于$\min\{a,b\}$，且接受概率位于$[0,1]$。因此该上界可以达到。将$a,b$换回式\eqref{eq:V5-C03-directed-flow}即得MH接受率。
\end{proof}

\paragraph{常见特例。}
若$q(x,y)=q(y,x)$，提议项消去，
\[
\alpha(x,y)=\min\{1,\widetilde\pi(y)/\widetilde\pi(x)\}.
\]
若$q(x,y)=q(y)$与当前状态无关，则
\[
\alpha(x,y)=\min\!\left\{1,
\frac{\widetilde\pi(y)q(x)}
{\widetilde\pi(x)q(y)}\right\}.
\]
对数实现使用
\[
\Delta=\log\widetilde\pi(y)-\log\widetilde\pi(x)
+\log q(y,x)-\log q(x,y),
\qquad \log\alpha=\min\{0,\Delta\},
\]
但只有各正密度项可取对数；零分子必须进入确定拒绝分支。

\paragraph{单分量MH。}
写$x=(x_j,x_{-j})$，在保持$x_{-j}$不变时从$q_j(x_j,y_j\mid x_{-j})$提议$y_j$。令$y=(y_j,x_{-j})$，则
\begin{equation}\label{eq:V5-C03-component-alpha}
\alpha_j(x,y)
=\min\!\left\{1,
\frac{\widetilde\pi(y_j,x_{-j})
q_j(y_j,x_j\mid x_{-j})}
{\widetilde\pi(x_j,x_{-j})
q_j(x_j,y_j\mid x_{-j})}\right\}
\end{equation}
在分母正时成立，零流仍按式\eqref{eq:V5-C03-common-flow}处理。

\begin{theorem}[有限状态MH的正确性条件]\label{thm:V5-C03-finite-correctness}
设$\mathcal X$有限，$\pi(x)>0$对每个$x\in\mathcal X$成立，$K$为以$\pi$为目标的MH核。
\begin{enumerate}[leftmargin=2.2em]
  \item $\pi$是$K$的平稳分布。
  \item 若$K$不可约，则$\pi$是唯一平稳分布，且对任意$f:\mathcal X\to\mathbb R$及任意初值，
  \[
  \frac1n\sum_{t=1}^nf(X_t)\longrightarrow E_\pi f
  \quad\text{几乎必然}.
  \]
  \item 若$K$还非周期，则对任意初始分布$\nu_0$都有$\nu_0K^t\to\pi$。
\end{enumerate}
\end{theorem}
% KNOWLEDGE-PLAN: KN-V5-C32-PROOF-006 L1
\paragraph{读前自检：证明：有限状态MH的正确性条件。}有限状态MH已知不可约且以$\pi$可逆；请把$f(X_t)$展开为$\sum_x f(x)\boldsymbol1_{\{X_t=x\}}$并交换有限求和与时间极限，核对时间平均只需不可约，而$\nu_0K^t\to\pi$还需非周期性。\par

\begin{proof}
第1项由定理\ref{thm:V5-C03-mh-reversible}得到。定理\ref{thm:V5-C01-finite-stationary}在同一行随机与平稳约定下给出：有限不可约链的平稳分布唯一且各状态正常返。再由定理\ref{thm:V5-C01-ergodic}的式\eqref{eq:V5-C01-time-average}，把$f$写成有限和
\[
f(X_t)=\sum_{x\in\mathcal X}f(x)\boldsymbol1_{\{X_t=x\}},
\]
对每个状态的访问频率使用该遍历定理，再交换有限求和与极限，得到第2项。若再加非周期性，定理\ref{thm:V5-C01-ergodic}的式\eqref{eq:V5-C01-step-convergence}给出$K^t(x,\cdot)\to\pi$；按初始分布$\nu_0$作有限加权和即得$\nu_0K^t\to\pi$。非周期性没有用于时间平均结论，因此不能把第2项与第3项合并成一个无条件断言。
\end{proof}

定理中的时间平均也覆盖任意固定燃烧期$m$。事实上，若
$S_N=N^{-1}\sum_{t=1}^Nf(X_t)\to I$，则
\[
\frac1n\sum_{t=m+1}^{m+n}f(X_t)
=\frac{m+n}{n}S_{m+n}
-\frac1n\sum_{t=1}^mf(X_t)
\longrightarrow I,
\]
因为$m$固定，第二项的分子只含有限项。若$X_0\sim\pi$，则每个$X_t\sim\pi$，并且
\[
E_\pi(\widehat I_{m,n})
=\frac1n\sum_{t=m+1}^{m+n}E_\pi f(X_t)
=E_\pi f.
\]
若从固定$x$启动，
\[
E_x(\widehat I_{m,n})
=\frac1n\sum_{t=m+1}^{m+n}K^tf(x),
\]
有限$n$时一般不等于$E_\pi f$。所以固定燃烧期不保证有限样本无偏；它只在已有收敛条件下影响初值偏差的大小。

\paragraph{读前自检：一般状态空间的可达与回返条件。}一般状态空间的$\psi$-不可约性要求每个$\psi(A)>0$的集合都能从任意初态以正概率到达；请给定一个集合$A$检查$P_x(\tau_A<\infty)>0$，并与概率为1的回返条件区分。\par

\begin{definition}[一般状态空间的可达与回返条件]\label{def:V5-C03-harris-conditions}
给定非零$\sigma$有限测度$\psi$，若每个满足$\psi(A)>0$的可测集$A$都满足$P_x(\tau_A<\infty)>0$（对所有$x$），则称链为$\psi$-不可约；若进一步$P_x(\tau_A<\infty)=1$（对所有$x$），则称其Harris回返。Harris回返链若存在不变概率分布$\pi$，称为正常Harris返。这里的陈述针对集合而不是单点，所以适用于连续状态空间。
\end{definition}

\begin{proposition}[一般状态空间结论的条件分层]\label{prop:V5-C03-general-convergence}
设$K$为正常Harris返核且不变概率分布为$\pi$。
\begin{enumerate}[leftmargin=2.2em]
  \item 对每个$f\in L^1(\pi)$，时间平均$n^{-1}\sum_{t=1}^nf(X_t)$几乎必然收敛到$E_\pi f$；该结论不要求非周期性。
  \item 若$K$还非周期，则对该遍历定理覆盖的初始状态有$\lVert K^t(x,\cdot)-\pi\rVert_{\mathrm{TV}}\to0$，从而固定时刻的边缘分布收敛；这不是时间平均结论的同义改写。
  \item 中心极限定理、有限渐近方差以及批均值估计的一致性都不是前两项的自动推论；它们还需相应矩条件与混合条件，例如几何遍历配合$E_\pi|f|^{2+\delta}<\infty$，并需按所用批均值定理选择批长增长条件。
\end{enumerate}
\end{proposition}

\begin{comparisonbox}
\textbf{三个反例把结论边界分开。}
两点确定翻转链不可约且具有唯一平稳分布$(1/2,1/2)$，状态访问频率与可积函数时间平均仍正确，但固定时刻分布按奇偶振荡；缺少的是非周期性。恒等核$I_2$可逆且保持任意分布，却可约、平稳分布不唯一，时间平均由初值决定；缺少的是跨类可达性。整数集$\mathbb Z$上的简单对称随机游走不可约且Harris回返，但为零常返，没有不变概率分布；因此不能把“会返回”替换成正常Harris返。最后，即使正常Harris返保证$L^1$大数律，若$f$没有所需二阶矩或链缺少相应混合速率，中心极限定理与批均值误差估计仍可能失效。
\end{comparisonbox}

因此，连续状态空间中“从任意点到任意点的概率为正”既不必要也通常不可验证。平稳性只说明目标保持；$\psi$-不可约性只说明正$\psi$测度集合可达；正常Harris返才支撑$L^1$时间平均；非周期性另用于总变差或边缘收敛；MCSE与批均值还要额外的矩和混合假设。有限状态定理不能只靠把求和符号改成积分就自动推广。

\begin{proposition}[平稳相关均值的有限样本方差]\label{prop:V5-C03-correlated-variance}
设链从平稳分布启动，$Y_t=f(X_t)$具有有限方差$\gamma_0>0$，并令
\[
\gamma_k=\operatorname{Cov}(Y_t,Y_{t+k}),
\qquad \rho_k=\gamma_k/\gamma_0.
\]
则对$\overline Y_n=n^{-1}\sum_{t=1}^nY_t$，
\begin{equation}\label{eq:V5-C03-finite-variance}
\operatorname{Var}(\overline Y_n)
=\frac{\gamma_0}{n}
\left[
1+2\sum_{k=1}^{n-1}
\left(1-\frac{k}{n}\right)\rho_k
\right].
\end{equation}
\end{proposition}
% KNOWLEDGE-PLAN: KN-V5-C32-PROOF-007 L1
\paragraph{读前自检：证明：平稳相关均值的有限样本方差。}在“$\operatorname{Var}(\overline Y_n) =\frac1{n^2}\left[n\gamma_0+ 2\sum_{k=1}^{n-1}(n-k)\gamma_k\right].$”成立时处理平稳相关均值的有限样本方差：请对$\operatorname{Var}(\overline Y_n)$做一次条件化或边缘化，并以所得量与证明紧接着给出的概率关系相同判定结果。\par

\begin{proof}
直接展开协方差双重和：
\[
\operatorname{Var}(\overline Y_n)
=\frac1{n^2}\sum_{s=1}^n\sum_{t=1}^n
\operatorname{Cov}(Y_s,Y_t).
\]
平稳性使协方差只依赖$|s-t|$。滞后$0$有$n$项；对每个$1\le k\le n-1$，上三角和下三角各有$n-k$对相距$k$的时刻。因此
\[
\operatorname{Var}(\overline Y_n)
=\frac1{n^2}\left[n\gamma_0+
2\sum_{k=1}^{n-1}(n-k)\gamma_k\right].
\]
提出$\gamma_0/n$并使用$\rho_k=\gamma_k/\gamma_0$，得到式\eqref{eq:V5-C03-finite-variance}。
\end{proof}

若$\sum_{k\ge1}|\rho_k|<\infty$，定义
\begin{equation}\label{eq:V5-C03-iact}
\tau_{\mathrm{int}}
=1+2\sum_{k=1}^{\infty}\rho_k,
\qquad
\sigma_{\mathrm{as}}^2=\gamma_0\tau_{\mathrm{int}}.
\end{equation}
在另有适当的马尔可夫链中心极限定理条件时，
$\sqrt n(\overline Y_n-E_\pi f)$渐近服从方差$\sigma_{\mathrm{as}}^2$的正态分布。有限样本的方差等效有效样本量为
\begin{equation}\label{eq:V5-C03-finite-ess}
N_{\mathrm{eff},n}
=\frac{n}{
1+2\sum_{k=1}^{n-1}(1-k/n)\rho_k},
\end{equation}
前提是分母正且使用真实自相关。用截断经验自相关替换它只能得到诊断估计；负相关时ESS可以大于$n$，因此不应机械裁剪到$n$。

批均值提供另一条渐近方差估计路径。令$n=ab$，把轨迹分成$a\ge2$个连续批、每批长度$b\ge1$，批均值为$\overline Y^{(j)}$，则

=\frac{b}{a-1}\sum_{j=1}^{a}
\bigl(\overline Y^{(j)}-\overline Y_n\bigr)^2,
\quad
\widehat{\mathrm{MCSE}}_{\mathrm{BM}}
=\sqrt{\widehat\sigma_{\mathrm{BM}}^2/n},
\quad
\widehat N_{\mathrm{eff,BM}}
=n\,s_Y^2/\widehat\sigma_{\mathrm{BM}}^2,
\end{equation}
最后一个比值只在分子、分母有限且分母正时定义。批均值一致性要求随总长度增长时$a,b$都增长且满足相应混合与矩条件；一条有限轨迹不能单独证明这些条件。

\textbf{优化解释。}命题\ref{prop:V5-C03-maximal-flow}说明，在逐对满足细致平衡且接受概率不超过$1$的限制下，MH选择最大的共同流。这是局部接受流最大化，不等于全局最小化自相关或MCSE。改变提议$q$会同时改变移动距离、接受概率和计算成本；只最大化接受率通常把步长推向过小。

\begin{example}[单向提议造成的支持连通缺口]\label{exm:V5-C03-asymmetric-proposal}
在三点空间上令目标分布为$\pi=(1/3,1/3,1/3)$，提议矩阵为定向三环
\[
Q=\begin{pmatrix}
0&1&0\\
0&0&1\\
1&0&0
\end{pmatrix}.
\]
计算MH接受率与所得核，并解释为什么“提议图从每点都有出边”仍不足以保证可用的MH链。
\end{example}

\SLExampleSolutionHeading{exm:V5-C03-asymmetric-proposal}
\begin{solution}
\SLReadTranslation{虽然定向三环从每个状态都有一条出边，但MH接受率还需要反向提议概率；要逐边计算接受率，写出完整核并判断链是否可用。}
\SolGiven 目标在三个状态上均为正且相等；$Q$ 按行给出提议概率，每条非对角转移为 $q_{ij}\alpha(i,j)$，拒绝概率补到对角线。
\SLMethodTrigger{MH比值含 $\pi_jq_{ji}/(\pi_iq_{ij})$；若正向边的反向提议为零，则该正向提议必被拒绝。}
\SolPlan 对 $1\to2$、$2\to3$、$3\to1$ 分别检查反向概率，得到全部接受率并补全 $K$；再区分“$\pi$平稳”和“链不可约并收敛”。
\SolDerive
\textbf{计算。}状态支持集是$S=\{1,2,3\}$，且每行提议概率之和为$1$。从$1$只能提议$2$，但反向$q_{21}=0$，故
\[
\alpha(1,2)=0.
\]
同理$q_{32}=q_{13}=0$分别使$\alpha(2,3)=\alpha(3,1)=0$，因而所有提议都被拒绝，所得核是$K=I_3$。

\SolCheck 虽然$\pi K=\pi$，但从点质量$\delta_1$出发有$\delta_1K^t=\delta_1$对所有$t$成立，不会趋近$\pi$；这直接证实平稳性没有带来不可约性。

\SolAnswer 三条提议边的接受率全为$0$，所得MH核为 $K=I_3$；任意分布虽都平稳，但链完全可约且不会从一般初值收敛到$\pi$。问题在于每条有向提议边都缺少反向流；后文例\ref{exm:V5-C03-three-state-kernel}再审计双向非对称提议。
\end{solution}

本模块给出三个可独立执行的过程：一般MH、单分量MH和轨迹诊断。随机数序列给定后，算法的控制路径完全确定；若中途停止，只使用此前完成的整次转移，不能把部分轨迹当成完整结果。

\phantomsection\label{struct:V5-C03-CH20}
\textbf{输入与输出。}一般MH输入可归一化的未归一化目标$\widetilde\pi$（满足$0<\int\widetilde\pi\,\mathrm d\mu<\infty$）、提议抽样器与双向密度求值器、合法初值$x_0$、相互独立的$(0,1)$均匀数、燃烧期$m$、保留数$n$及可选函数$f$。输出完整状态轨迹、提议、接受判定、保留样本，以及在$n>0$且$f$合法时的均值。

\phantomsection\label{struct:V5-C03-CH21}
\textbf{参数含义。}$T=m+n$是固定转移预算；$m$只决定哪些状态不进入均值，$n$决定保留状态数。提议尺度属于$q$的一部分。若采用单分量算法，还需维数$d$、每坐标提议$q_j$与扫描规则。诊断算法另需滞后上限$L$和满足$n=ab$的批数$a$、批长$b$。

\phantomsection\label{struct:V5-C03-CH22}
\textbf{初始化。}先确认目标可归一化，检查$0<\widetilde\pi(x_0)<\infty$和整数参数；令当前状态$x\leftarrow x_0$，轨迹以$x_0$开头，接受计数为$0$。不得从目标支持外启动后再靠一次非法比值“跳回”支持。

\phantomsection\label{struct:V5-C03-CH23}
\textbf{更新顺序。}每轮先由$q(x,\cdot)$提议$y$，再独立取$u\in(0,1)$；核对前向提议密度为有限正数。若$\widetilde\pi(y)=0$或$q(y,x)=0$，确定拒绝；否则计算对数比$\Delta$并比较$\log u$与$\min\{0,\Delta\}$。接受或拒绝后都必须追加新的链状态。

\phantomsection\label{struct:V5-C03-CH24}
\textbf{停止条件。}正常分支恰好执行$T$次转移。$T=0$时不调用任何随机数，返回只含$x_0$的轨迹和空保留集。遇到不是有限实数的目标或提议求值、负密度、正反提议求值不一致、非法均匀数或函数值异常时立即停止，只保留此前完成的整次转移。

\phantomsection\label{struct:V5-C03-CH25}
\textbf{正确性与收敛条件。}定理\ref{thm:V5-C03-mh-reversible}保证合法MH核保持$\pi$；定理\ref{thm:V5-C03-finite-correctness}说明有限状态下还需不可约性，边缘逐步收敛还需非周期性。一般状态空间必须按定义\ref{def:V5-C03-harris-conditions}与命题\ref{prop:V5-C03-general-convergence}分别核验正常Harris返、非周期性以及MCSE所需的额外矩与混合条件。算法完成只表示预算执行完毕，不表示这些数学条件已经由数据诊断证明。

\textbf{不收敛或不适用情形。}目标不可归一化、提议诱导的链无法经多步到达目标支持中的各个正质量区域（因而可约），或连续状态链不满足所需Harris返性时，增加燃烧期与样本数都不能补救正确性；轨迹诊断算法也不适用于非有限输入或批数不足的序列。

\textbf{复杂度假设。}下述界把一次目标求值、正反提议密度求值与随机数调用的实际成本显式记入$C_{\mathrm{MH}}$或$C_j$，并假设一次状态复制为$O(d)$；不把目标密度、提议器或坐标更新无条件视为常数时间。

  output={最后合法基础统计、ACF/加权ESS、批均值MCSE/ESS、已检轨迹数$y_{\rm done}$、滞后数$k_{\rm done}$、批数$b_{\rm done}$、状态与最终诊断},

  loop={先有限扫描轨迹与$L+1$个滞后，再有限扫描$a$个连续批},

}

\begin{algorithm}[tbp]
\caption{有限样本权重与批均值并列的轨迹诊断}\label{alg:V5-C03-trace-diagnostics}
\KwIn{非恒定的有限实数轨迹$y_{1:n}$；$0\le L<n$；$a\ge2,b\ge1$且$n=ab$}
\KwOut{最后合法统计；$y_{\rm done},k_{\rm done},b_{\rm done}$；$\mathtt{status}$与最终诊断}
置$y_{\rm done}=k_{\rm done}=b_{\rm done}=0$；若$n,L,a,b$或批划分非法，则返回空统计、$\mathtt{invalid\_input}$及字段诊断\;
逐项核验$y_t$有限并增加$y_{\rm done}$；若非有限或轨迹恒定，则返回已验证前缀信息、$\mathtt{invalid\_input}$及位置/方差诊断\;
计算$\bar y\leftarrow n^{-1}\sum_ty_t$与$s_Y^2\leftarrow(n-1)^{-1}\sum_t(y_t-\bar y)^2$\;
若均值或方差非有限或方差非正，则返回最后合法输入诊断、$\mathtt{numerical\_failure}$及基础统计阶段诊断\;
\For{$k\leftarrow0,\ldots,L$}{
  令$\widehat\gamma_k\leftarrow(n-k)^{-1}\sum_{t=1}^{n-k}(y_t-\bar y)(y_{t+k}-\bar y)$，
  $\widehat\rho_k\leftarrow\widehat\gamma_k/\widehat\gamma_0$；若非有限则返回基础统计与已提交ACF、$\mathtt{numerical\_failure}$及滞后$k$诊断；否则原子追加并增加$k_{\rm done}$\;
}
令$\widehat\tau_L\leftarrow1+2\sum_{k=1}^L(1-k/n)\widehat\rho_k$\;
若$\widehat\tau_L\le0$或非有限，则返回基础统计/ACF、$\mathtt{numerical\_failure}$及截断时间诊断；否则令$\widehat{\mathrm{ESS}}_L=n/\widehat\tau_L$\;
\For{$j\leftarrow1,\ldots,a$}{
  计算第$j$个连续批均值；若非有限则返回已有统计、$\mathtt{numerical\_failure}$及批$j$诊断；否则原子追加并增加$b_{\rm done}$\;
}
令$\widehat\sigma_{\mathrm{BM}}^2\leftarrow b(a-1)^{-1}\sum_j(\bar y^{(j)}-\bar y)^2$\;
$\widehat{\mathrm{MCSE}}_{\mathrm{BM}}\leftarrow\sqrt{\widehat\sigma_{\mathrm{BM}}^2/n}$并形成批均值ESS；若结果非有限或定义域失败，则返回前述统计、$\mathtt{numerical\_failure}$及批聚合诊断\;
原子提交完整诊断输出，返回实际计数、$\mathtt{completed}$及“有限轨迹诊断不是收敛证明”最终诊断\;
\end{algorithm}
\end{AlgorithmContract}

\paragraph{读前自检：复杂度分析：自相关与批均值诊断。}长度$T$的MCMC轨迹每步提议成本为$C_{\rm MH}$时，生成成本为$\Theta(TC_{\rm MH})$；请再加滞后$L$的自相关诊断成本$\Theta(TL)$，核对只在线计数与保存完整轨迹的空间分别不同。\par

\begin{complexitybox}
\textbf{时间复杂度。}设一次提议抽样、目标求值、双向提议密度求值的成本合计为$C_{\mathrm{MH}}$，状态复制成本为$O(d)$。一般MH执行$T=m+n$步的时间为$\Theta(T(C_{\mathrm{MH}}+d))$。单分量算法若第$j$个坐标更新成本为$C_j$，则$T$轮成本为$\Theta(T\sum_{j=1}^dC_j)$。直接计算$L$个经验自相关需$\Theta(nL)$，批均值与样本方差需$\Theta(n)$；这些界不把目标密度求值无条件视为常数。

\phantomsection\label{struct:V5-C03-CH27}
\textbf{空间复杂度。}只保留当前状态与在线接受计数时，一般MH工作空间为$\Theta(d)$；返回完整轨迹需$\Theta((m+n)d)$输出空间。单分量算法若保存轮内状态，额外输出可达$\Theta((m+n)d^2)$，通常只保存轮末状态和$d$个接受判定。诊断算法保存输入轨迹与$L+1$个自相关为$\Theta(n+L)$，批均值另需$\Theta(a)$。
\end{complexitybox}

\phantomsection\label{struct:V5-C03-CH28}
\begin{applicabilitybox}
\textbf{适用条件。}MH适用于能计算未归一化目标点值、能从提议核抽样并能计算正反提议密度的问题。目标必须可归一化，初值必须在正目标支持内，所得核还需在目标支持上具备足够可达性。单分量MH适合全维提议难设计、而条件坐标提议容易构造的高维问题。

\phantomsection\label{struct:V5-C03-CH29}
\textbf{局限性。}高维、多峰、强相关或狭窄支持会导致慢混合；密度比只使用局部点值，不能自动发现未访问模态。固定预算不能保证预定误差，有限轨迹也不能验证所有渐近条件。系统扫描虽保持目标，一般不再可逆；若目标本身不可归一化，任何调参都不能修复。

\phantomsection\label{struct:V5-C03-CH30}
\textbf{常见错误。}包括遗漏$q(y,x)/q(x,y)$；在$0/0$处硬算比值；删除拒绝产生的重复状态；把高接受率当作高ESS；用燃烧期修复可约性；把抽稀后样本称为独立；只画单链轨迹便宣布收敛；把细致平衡扩张为唯一性；对不可归一化核运行MCMC；以及报告未实际运行得到的随机样本或数值。

\phantomsection\label{struct:V5-C03-CH31}
\textbf{易混淆概念对比。}平稳性是$\pi K=\pi$，可逆性是更强的双向流条件；不可约性负责连接支持，非周期性负责消除边缘分布振荡。接受--拒绝抽样只输出接受候选并可产生iid样本，MH拒绝时保留当前状态并产生相关链。燃烧期丢弃前缀，抽稀选择子序列，二者都不同于证明收敛。MCSE量化给定诊断模型下的均值不确定性，ESS把相关方差换算成iid尺度，二者都不是核正确性的证明。
\end{applicabilitybox}

\begin{example}[由非对称提议构造完整MH核]\label{exm:V5-C03-three-state-kernel}
在状态空间$\{1,2,3\}$上，目标分布和提议矩阵为
\[
\pi=\left(\frac12,\frac13,\frac16\right),
\qquad
Q=\begin{pmatrix}
1/2&1/2&0\\
1/4&1/2&1/4\\
0&1/2&1/2
\end{pmatrix}.
\]
任务如下：逐条计算非对角接受率，写出包含拒绝自环的MH矩阵$K$；用双向流核验$\pi$平稳；判断不可约性与非周期性；最后从$x_0=1$出发，把固定测试输入
\[
(y_1,u_1)=(2,0.2),\qquad (y_2,u_2)=(3,0.8)
\]
送入算法，给出两次转移后的状态。这里的两个数是确定性测试输入，不声称来自一次真实随机运行。
\end{example}

\SLExampleSolutionHeading{exm:V5-C03-three-state-kernel}
\begin{solution}
\SLReadTranslation{要把非对称提议逐边校正成完整MH核：计算双向接受率，把拒绝质量补到对角线，再核验平稳性、连通性、周期和两次固定输入分支。}
\SolGiven $\pi_i>0$ 且 $Q$ 行随机；接受率为 $\alpha_{ij}=\min\{1,\pi_jq_{ji}/(\pi_iq_{ij})\}$，非对角核元为 $q_{ij}\alpha_{ij}$。
\SLMethodTrigger{逐对比较正反概率流可同时得到接受率和细致平衡；所有非对角元确定后，对角元必须由每行和为$1$唯一补足。}
\SolPlan 分别处理边 $1\leftrightarrow2$、$2\leftrightarrow3$，构造完整 $K$；核对双向流、连通图和自环，再按给定 $(y_t,u_t)$ 顺序执行两次接受判定。
\SolDerive
\textbf{接受率。}对边$1\leftrightarrow2$，
\[
\alpha_{12}
=\min\left\{1,\frac{(1/3)(1/4)}{(1/2)(1/2)}\right\}
=\frac13,
\qquad
\alpha_{21}=1.
\]
对边$2\leftrightarrow3$，
\[
\alpha_{23}
=\min\left\{1,\frac{(1/6)(1/2)}{(1/3)(1/4)}\right\}=1,
\qquad
\alpha_{32}=1.
\]
$q_{13}=q_{31}=0$，这对边不产生提议流。\textbf{完整核。}非对角转移概率是$q_{ij}\alpha_{ij}$，对角元由行和为$1$补足，因此
\[
K=\begin{pmatrix}
5/6&1/6&0\\
1/4&1/2&1/4\\
0&1/2&1/2
\end{pmatrix}.
\]
\textbf{平稳性与结构。}两个非零无向边上的流分别满足
\[
\pi_1K_{12}=\frac12\cdot\frac16=\frac1{12}
=\frac13\cdot\frac14=\pi_2K_{21},
\]
\[
\pi_2K_{23}=\frac13\cdot\frac14=\frac1{12}
=\frac16\cdot\frac12=\pi_3K_{32}.
\]
对角流与自身相等，故细致平衡成立；按命题\ref{prop:V5-C03-db-stationary}，$\pi K=\pi$。图$1-2-3$连通，所以$K$不可约；三个状态都有正自环，因此周期为$1$。有限状态定理\ref{thm:V5-C03-finite-correctness}遂给出唯一平稳分布、时间平均收敛与边缘分布收敛。

\textbf{确定性分支测试。}第1步从$1$提议$2$，因为$u_1=0.2\le1/3$而接受，得到$x_1=2$；第2步从$2$提议$3$，接受概率为$1$，故$u_2=0.8$仍导致接受，得到$x_2=3$。

\SolCheck 除逐边流量检查外，直接相乘也有
\[
\pi K=\left(\frac12,\frac13,\frac16\right),
\qquad K\boldsymbol1=\boldsymbol1,
\]
分别复核平稳性与随机矩阵约束。

\SolAnswer 所得核为 $K=\begin{psmallmatrix}5/6&1/6&0\\1/4&1/2&1/4\\0&1/2&1/2\end{psmallmatrix}$，不可约、非周期并以$\pi$为唯一平稳分布；固定输入得到 $x_1=2,x_2=3$，只用于核验分支逻辑。
\SLStuckHint{先只写出对象、合法条件和第一轮/第一步中间量；若这些量无法定义，再返回相应定义。}
\end{solution}

从一般\MetropolisHastings{}算法推导单分量\MetropolisHastings{}算法。要求写出坐标提议核、正反提议流、接受概率和一轮系统扫描，并说明系统扫描保持目标分布但一般不再可逆。

\phantomsection\label{sol:V5-C03-S03-01}
\begin{chapterexercisesolution}{ex:V5-C03-S03-01}
设每个坐标空间$(\mathcal X_j,\mathcal B_j)$为标准Borel空间，$\mu_j$为$\sigma$有限测度，并要求$q_j(x_j,y_j\mid x_{-j})$关于$(x,y_j)$联合可测、非负，且对每个$x$满足$\int q_j(x_j,y_j\mid x_{-j})\mu_j(\mathrm dy_j)=1$。于是对任意可测矩形$B=B_j\times B_{-j}$，$x\mapsto Q_j(x,B)$可测；由单调类定理推广到所有乘积$\sigma$代数中的$B$，故下式确实定义马尔可夫核：
写$x=(x_j,x_{-j})$，单次只改第$j$个分量。坐标提议核可写为
\[
Q_j(x,\mathrm dy)
=q_j(x_j,y_j\mid x_{-j})\mu_j(\mathrm dy_j)
\delta_{x_{-j}}(\mathrm dy_{-j}).
\]
若$y=(y_j,x_{-j})$，正向和反向未归一化流分别是
\[
g_j(x,y)=\widetilde\pi(x_j,x_{-j})
q_j(x_j,y_j\mid x_{-j}),
\]
\[
g_j(y,x)=\widetilde\pi(y_j,x_{-j})
q_j(y_j,x_j\mid x_{-j}).
\]
取公共流$\min\{g_j(x,y),g_j(y,x)\}$并除以正向流，得到
\[
\alpha_j(x,y)
=\min\!\left\{1,
\frac{\widetilde\pi(y_j,x_{-j})
q_j(y_j,x_j\mid x_{-j})}
{\widetilde\pi(x_j,x_{-j})
q_j(x_j,y_j\mid x_{-j})}\right\},
\]
这正是式\eqref{eq:V5-C03-component-alpha}。分母为零时不计算比值，而按零流分支处理。该坐标核的接受流对称，拒绝质量形成自环，故关于$\pi$可逆并保持$\pi$。

一轮系统扫描从当前状态开始，按$j=1,\ldots,d$依次提议、判定并立即更新；后一个坐标看到的是前一个坐标更新后的状态。若轮核记为$K_{\mathrm{sys}}=K_1\cdots K_d$，则
\[
\pi K_{\mathrm{sys}}
=((\pi K_1)K_2\cdots)K_d=\pi.
\]
但伴随核的顺序是$K_d\cdots K_1$；这些核不交换时，正反顺序不同，系统扫描一般不再可逆。随机扫描$\sum_jw_jK_j$是可逆核的凸组合，因而自动继承可逆性。
\end{chapterexercisesolution}

先验密度在$(0,1)$上为$\theta^{1-1}(1-\theta)^{1-1}/B(1,1)$。观测$10$次、其中成功$4$次时，完整二项似然为

\theta\mid y\sim\operatorname{Beta}(1+4,1+10-4)
=\operatorname{Beta}(5,7).
\]

\begin{equation}\label{eq:V5-C03-beta-benchmark}
E(\theta\mid y)=\frac5{12},
\qquad
\operatorname{Var}(\theta\mid y)
=\frac{5\cdot7}{(5+7)^2(5+7+1)}
=\frac{35}{1872}\approx0.0186966.
\end{equation}

\widetilde\pi(\theta)=\theta^4(1-\theta)^6\boldsymbol1_{(0,1)}(\theta),
\qquad q(x,y)=\boldsymbol1_{(0,1)}(y).
\]

\[
\alpha(x,y)
=\min\left\{1,
\frac{y^4(1-y)^6}{x^4(1-x)^6}\right\}.
\]

\[
\widehat\mu=\frac1n\sum_{t=m+1}^{m+n}X_t,
\qquad
\widehat v=\frac1n\sum_{t=m+1}^{m+n}(X_t-\widehat\mu)^2.
\]

\begin{exercise}\label{ex:V5-C03-S04-01}

\ifSLFullEdition
\phantomsection\label{sol:V5-C03-S04-01}
\begin{chapterexercisesolution}{ex:V5-C03-S04-01}
Barker正向流为
\[
a\alpha_{\mathrm B}(x,y)
=a\frac{b}{a+b}
=\frac{ab}{a+b}.
\]
反向接受率是$a/(a+b)$，所以反向流
\[
b\alpha_{\mathrm B}(y,x)
=b\frac{a}{a+b}
=\frac{ab}{a+b},
\]

\[
\frac{ab}{a+b}\le\frac{ab}{b}=a=\min\{a,b\};
\]

u\notin(0,1)&:\quad \text{随机源非法，当前转移不提交}.
\end{aligned}
\]
其中$\widetilde\pi(y)=0$或$q(y,x)=0$是合法零反向流，必须把拒绝后的重复状态追加到轨迹；它们不同于$q(x,y)=0$却被抽到的一致性错误。若$n=1$，仅有
\[
\widehat\mu=X_{m+1},\qquad
s^2=\frac{\sum_{t=m+1}^{m+n}(X_t-\widehat\mu)^2}{n-1}
\quad\text{未定义},
\]
故样本方差、自相关与ESS均标记未定义，不得除零。
\end{chapterexercisesolution}

\begin{exercise}\label{ex:V5-C03-S06-04}
从输出对象、全局包络、拒绝后的行为、样本独立性、归一化常数与不能继续的情形六个方面比较接受--拒绝抽样和MH。说明为什么不能把接受--拒绝法中的接受概率$p(y)/(cq(y))$直接替换成MH接受概率而仍沿用原输出规则。
\end{exercise}
\ifSLFullEdition
\phantomsection\label{sol:V5-C03-S06-04}
\begin{chapterexercisesolution}{ex:V5-C03-S06-04}
\begin{center}
\small
\begin{tabularx}{\linewidth}{@{}l X X@{}}
\toprule 比较项 & 接受--拒绝抽样 & \MetropolisHastings{}\\ \midrule
输出对象 & 只输出接受候选 & 每轮输出下一链状态，拒绝时重复当前状态\\
全局条件 & 需要$p(y)\le c q(y)$的有限包络 & 不需要全局包络，但需要合法双向提议流\\
拒绝后行为 & 丢弃候选并从$q$重新独立提议 & 保留当前状态，形成自环\\
独立性 & 独立提议下，接受值为iid目标样本 & 一般为相关马尔可夫链\\
归一化常数 & 接受概率通常使用归一化目标与已知包络 & 目标公共有限正常数在双向比值中消去\\
不能继续的情形 & 包络越界、达到预设提议上限与提议异常 & 目标不可归一化、支持断裂、双向提议密度不合法或未完成预设转移数\\
\bottomrule
\end{tabularx}
\end{center}
接受--拒绝的条件输出分布依赖全局包络归一化：
\[
P(Y\in\mathrm dy\mid A)
=\frac{q(y)\,p(y)/(cq(y))}{P(A)}\,\mathrm dy
=\frac{p(y)/c}{1/c}\,\mathrm dy=p(y)\,\mathrm dy.
\]
MH则定义完整状态核
\[
K(x,\mathrm dy)=q(x,y)\alpha(x,y)\,\mathrm dy
+\left[1-\int q(x,z)\alpha(x,z)\,\mathrm dz\right]\delta_x(\mathrm dy).
\]
若把接受--拒绝概率换成MH接受率却仍丢弃拒绝并只收集接受候选，条件密度变为
\[
\widetilde p_x(y)
=\frac{q(x,y)\alpha(x,y)}
{\int q(x,z)\alpha(x,z)\,\mathrm dz},
\]
它一般依赖当前$x$且不等于$p(y)$；同时MH核中的拒绝自环消失。反过来，把iid接受值按MH轨迹解释也会凭空引入错误依赖。算法必须连同输出规则整体使用。
\end{chapterexercisesolution}

\begin{exercise}\label{ex:V5-C03-S05-05}
计划运行四条不同初值的链。写出一份不依赖具体程序语言的诊断报告清单，至少包括完整未抽稀轨迹、运行均值、接受标志、经验自相关、批均值MCSE、ESS、链间比较和异常状态。逐项说明它能发现什么，以及不能证明什么。
\end{exercise}
\ifSLFullEdition
\phantomsection\label{sol:V5-C03-S05-05}
\begin{chapterexercisesolution}{ex:V5-C03-S05-05}
一份合格报告至少包含：
\begin{enumerate}[leftmargin=2.2em]
  \item \textbf{完整未抽稀轨迹：}可发现长时间停滞、边界粘连和不同模态；不能证明未画出的区域没有问题。
  \item \textbf{运行均值：}可显示估计是否仍有系统漂移；平台期不能证明平台等于目标期望。
  \item \textbf{提议与接受标志：}可定位零反向流、连续拒绝和数值异常；某个经验接受率不能证明可达性。
  \item \textbf{经验自相关：}可量化已访问轨迹的滞后依赖并构造截断ESS；滞后上限和估计噪声会影响结果。
  \item \textbf{批均值MCSE与ESS：}可把均值波动换算为误差尺度；短批、少批或慢混合会使估计不可靠。
  \item \textbf{四链比较：}不同初值链分离可暴露多模态和初值敏感；四链重合仍不能形式证明已经覆盖全部目标支持。
  \item \textbf{异常与停止原因：}记录不是有限实数的求值、零目标、零反向提议、达到预设转移上限和已经完成的转移；没有异常不等于数学条件已经证明。
\end{enumerate}
报告中的量必须能由保存的状态递推复算：
\[
\begin{aligned}
X_t&=\begin{cases}Y_t,&U_t\le\alpha_t,\\X_{t-1},&U_t>\alpha_t,\end{cases}\\
\bar f_t&=\frac1t\sum_{s=1}^tf(X_s),\\
\widehat a_t&=\frac1t\sum_{s=1}^tA_s,
\qquad A_s=\boldsymbol1\{U_s\le\alpha_s\},\\
\widehat{\operatorname{MCSE}}_{\rm BM}
&=\sqrt{\widehat\sigma_{\rm BM}^2/n},\qquad
\widehat N_{\rm eff}=n s_f^2/\widehat\sigma_{\rm BM}^2.
\end{aligned}
\]
四链比较还应保留逐链估计而非只保留合并均值：
\[
\widehat\mu_c=\frac1n\sum_{t=1}^nf(X_t^{(c)}),\qquad c=1,2,3,4;\qquad
\max_c\widehat\mu_c-\min_c\widehat\mu_c\text{大}\Longrightarrow\text{链间分离警报}.
\]
最后两式仅在相应分母有限且为正时报告。还应列出目标分布的具体定义、提议参数、$m,n$、批划分、随机种子及生成规则，使诊断可复算；停止时报告已完成的整次转移数和原因。结论措辞应是“未发现某类问题”或“出现某类警报”，不能写成“诊断证明收敛”。
\end{chapterexercisesolution}
