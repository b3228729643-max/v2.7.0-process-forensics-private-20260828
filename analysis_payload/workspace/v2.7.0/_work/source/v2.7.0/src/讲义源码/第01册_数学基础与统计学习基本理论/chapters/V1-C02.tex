\subsection*{问题提出}\label{schema:V1-C02:CH-06}
数据表、线性预测、最小二乘、协方差和降维都依赖矩阵。核心问题不是记住运算表，而是判断：线性映射保留了哪些方向，丢失了哪些方向；一个向量如何分解为子空间内分量和正交残差；迭代为何会突出主特征方向。

\begin{definition}[线性映射]\label{def:V1-C02-linear-map}\label{schema:V1-C02:CH-07}
设$T:\R^n\to\R^m$。若对任意$\boldsymbol{x},\boldsymbol{y}\in\R^n$和$a,b\in\R$都有
$T(a\boldsymbol{x}+b\boldsymbol{y})=aT(\boldsymbol{x})+bT(\boldsymbol{y})$，则称$T$为线性映射。给定标准基后，存在唯一$A\in\R^{m\times n}$使$T(\boldsymbol{x})=A\boldsymbol{x}$。
\end{definition}

\subsection*{模型、策略与算法}
\phantomsection\label{schema:V1-C02:CH-08}
本章把矩阵$A$视为线性模型：输入$\boldsymbol{x}\in\R^n$，输出$A\boldsymbol{x}\in\R^m$；列空间描述可达到的输出，零空间描述映射为零的输入。

\phantomsection\label{schema:V1-C02:CH-09}
学习策略是先按分量核对运算，再用子空间解释结构，最后用正交性或谱性质寻找最简表示。对近似问题，选择使欧氏残差最小的投影；对实对称矩阵，Rayleigh商最大化给出最大代数特征值的方向。只有协方差等半正定矩阵中，这一方向才同时是最大方差方向。

\phantomsection\label{schema:V1-C02:CH-10}
本章学习算法采用归一化幂法：反复施加矩阵并归一化，使初始向量中占优的特征方向相对放大。算法的完整输入、更新与终止规则见算法\ref{alg:V1-C02-power}。

\subsection*{核心推导：一维正交投影系数}
% CORE-DERIVATION-PLAN: KN-V1-C02-DERIVATION-001
\SLKnowledgeBridge{把“到直线最近”化为一个标量二次函数的极小化。}{$\boldsymbol{x},\boldsymbol{u}\in\R^n$ 与候选投影点 $a\boldsymbol{u}$。}{$\boldsymbol{u}\neq\boldsymbol0$，故 $\boldsymbol{u}^{\mathsf T}\boldsymbol{u}>0$；$a\in\R$。}{先展开 $g(a)=\norm{\boldsymbol{x}-a\boldsymbol{u}}_2^2$，再对标量 $a$ 求导。}

\begin{derivationgoalbox}\label{schema:V1-C02:CH-11}
给定非零$\boldsymbol{u}\in\R^n$和$\boldsymbol{x}\in\R^n$，求$\boldsymbol{x}$在$\operatorname{span}(\boldsymbol{u})$上的最近点$a\boldsymbol{u}$，并得到系数$a$。
\end{derivationgoalbox}
\begin{derivationprepbox}
目标函数为$g(a)=\norm{\boldsymbol{x}-a\boldsymbol{u}}_2^2$，定义域为$a\in\R$。因为$\boldsymbol{u}\ne\boldsymbol0$，有$\boldsymbol{u}^{\mathsf T}\boldsymbol{u}>0$。
\end{derivationprepbox}
% DERIVATION-CHECK: step-1; step-2; step-3
\textbf{推导第1步：展开平方。} 使用内积的双线性与对称性，
\[
g(a)=\boldsymbol{x}^{\mathsf T}\boldsymbol{x}-2a\boldsymbol{u}^{\mathsf T}\boldsymbol{x}
+a^2\boldsymbol{u}^{\mathsf T}\boldsymbol{u}.
\]
每一项都是标量，维数相容。

\textbf{推导第2步：求驻点。} 对标量$a$求导，
\[
g'(a)=-2\boldsymbol{u}^{\mathsf T}\boldsymbol{x}
+2a\boldsymbol{u}^{\mathsf T}\boldsymbol{u}.
\]
令$g'(a)=0$的依据是最小点位于无约束实数域内部且$g$可微。

\textbf{推导第3步：验证最小性。} 解得
\[
a^{\star}=\frac{\boldsymbol{u}^{\mathsf T}\boldsymbol{x}}
{\boldsymbol{u}^{\mathsf T}\boldsymbol{u}},\qquad
g''(a)=2\boldsymbol{u}^{\mathsf T}\boldsymbol{u}>0.
\]
二阶导数为正，所以$a^{\star}\boldsymbol{u}$是唯一最近点；残差与$\boldsymbol{u}$的内积为零。

\begin{theorem}[子空间正交投影的最佳逼近]\label{thm:V1-C02-best-projection}\label{schema:V1-C02:CH-12}
设$S$是$\R^n$的子空间，$\boldsymbol{x}=\boldsymbol{p}+\boldsymbol{r}$，其中$\boldsymbol{p}\in S$且$\boldsymbol{r}\in S^{\perp}$。则对任意$\boldsymbol{s}\in S$，
$\norm{\boldsymbol{x}-\boldsymbol{p}}_2\le\norm{\boldsymbol{x}-\boldsymbol{s}}_2$，等号成立当且仅当$\boldsymbol{s}=\boldsymbol{p}$。
\end{theorem}
% KNOWLEDGE-PLAN: KN-V1-C02-PROOF-001 L1
\paragraph{读前自检：证明：子空间正交投影的最佳逼近。}子空间最佳逼近证明给出$\boldsymbol{x}=\boldsymbol p+\boldsymbol r$，其中$\boldsymbol p,\boldsymbol s\in S$且$\boldsymbol r\in S^\perp$；请展开$\lVert\boldsymbol{x}-\boldsymbol s\rVert_2^2$，消去交叉项并核对等号仅在$\boldsymbol s=\boldsymbol p$时成立。\par

\begin{proof}\label{schema:V1-C02:CH-13}
证明目标是比较任意候选$\boldsymbol{s}$与$\boldsymbol{p}$的误差。由分解条件，
$\boldsymbol{x}-\boldsymbol{s}=\boldsymbol{r}+(\boldsymbol{p}-\boldsymbol{s})$。因为$\boldsymbol{p}-\boldsymbol{s}\in S$且$\boldsymbol{r}\in S^{\perp}$，两项内积为零。展开平方范数得到
\[
\norm{\boldsymbol{x}-\boldsymbol{s}}_2^2
=\norm{\boldsymbol{r}}_2^2+\norm{\boldsymbol{p}-\boldsymbol{s}}_2^2
\ge \norm{\boldsymbol{r}}_2^2
=\norm{\boldsymbol{x}-\boldsymbol{p}}_2^2.
\]
平方根在非负实数上保持次序。等号成立恰要求$\norm{\boldsymbol{p}-\boldsymbol{s}}_2^2=0$，也就是$\boldsymbol{s}=\boldsymbol{p}$，结论成立。
\end{proof}

\begin{geometrybox}\label{schema:V1-C02:CH-14}
\textbf{几何解释。} 投影把向量分成子空间内分量与垂直残差；特征向量是在线性变换下方向不变的轴，特征值给出该轴上的伸缩。秩是输出空间中独立方向的个数。
\end{geometrybox}
\begin{probabilitybox}\label{schema:V1-C02:CH-15}
随机向量中心化后的二阶矩形成协方差矩阵。任意方向$\boldsymbol{q}$上的方差写成二次型$\boldsymbol{q}^{\mathsf T}\Sigma\boldsymbol{q}$，所以最大方差方向与对称矩阵的主特征向量相连。
\end{probabilitybox}
\begin{optimizationbox}\label{schema:V1-C02:CH-16}
\textbf{优化解释。} 最小二乘是在列空间中寻找离观测向量最近的点；正规方程表达残差与列空间正交。对实对称矩阵，Rayleigh商最大化得到最大代数特征值的单位特征向量；它与“最大模特征值”不是同一概念。
\end{optimizationbox}

\input{../../绘图源码/第01册_数学基础与统计学习基本理论/V1-C02/fig_v1_c02_projection.tex}
由\cref{fig:V1-C02-projection}中的正交证书，对任意$\boldsymbol s\in S$都有
$\|\boldsymbol x-\boldsymbol s\|_2^2=\|(I-P_S)\boldsymbol x\|_2^2+\|P_S\boldsymbol x-\boldsymbol s\|_2^2$；第二项只在$\boldsymbol s=P_S\boldsymbol x$时为零，这就给出最短距离而非仅凭图形判断。

\begin{example}[二维投影与残差]\label{exm:V1-C02-projection}\label{schema:V1-C02:CH-18}\label{schema:V1-C02:CH-32}
令$\boldsymbol{x}=(3,1)^{\mathsf T}$，$\boldsymbol{u}=(1,1)^{\mathsf T}$。求$\boldsymbol{x}$到$\operatorname{span}(\boldsymbol{u})$的投影与距离。
\end{example}
\phantomsection\label{schema:V1-C02:CH-33}
\SLExampleSolutionHeading{exm:V1-C02-projection}
\begin{solution}
\SLReadTranslation{把$\boldsymbol x$正交投影到$\boldsymbol u$张成的直线上，求投影点、残差及点到直线的欧氏距离。}
\SolGiven $\boldsymbol x,\boldsymbol u\in\R^2$且$\boldsymbol u\ne\boldsymbol0$，故$\boldsymbol u^{\mathsf T}\boldsymbol u>0$，投影系数有定义。
\SLMethodTrigger{用一维子空间的正交投影公式求系数和投影，再由残差范数求距离；以正交性和勾股恒等式核验。}
\SolPlan 计算$a=(\boldsymbol u^{\mathsf T}\boldsymbol x)/(\boldsymbol u^{\mathsf T}\boldsymbol u)$，令$\boldsymbol p=a\boldsymbol u$、$\boldsymbol r=\boldsymbol x-\boldsymbol p$，最后求$\|\boldsymbol r\|_2$。
\SolDerive
对象与条件为$\boldsymbol{x},\boldsymbol{u}\in\R^2$且$\boldsymbol{u}\ne\boldsymbol0$，因此投影系数的分母严格为正。逐步计算
\[
\begin{aligned}
\boldsymbol{u}^{\mathsf T}\boldsymbol{x}&=1\cdot3+1\cdot1=4,
&\boldsymbol{u}^{\mathsf T}\boldsymbol{u}&=1^2+1^2=2,\\
a^\star&=\frac{\boldsymbol{u}^{\mathsf T}\boldsymbol{x}}
{\boldsymbol{u}^{\mathsf T}\boldsymbol{u}}=2,
&\boldsymbol{p}&=a^\star\boldsymbol{u}=(2,2)^{\mathsf T},\\
\boldsymbol{r}&=\boldsymbol{x}-\boldsymbol{p}=(1,-1)^{\mathsf T},
&\boldsymbol{u}^{\mathsf T}\boldsymbol{r}&=1-1=0,\\
\operatorname{dist}(\boldsymbol{x},\operatorname{span}(\boldsymbol{u}))
&=\norm{\boldsymbol r}_2=\sqrt{1^2+(-1)^2}=\sqrt2.
\end{aligned}
\]
\SolCheck $\boldsymbol u^{\mathsf T}\boldsymbol r=0$说明残差与子空间正交；另有$\norm{\boldsymbol x}_2^2=10=8+2=\norm{\boldsymbol p}_2^2+\norm{\boldsymbol r}_2^2$，两项证书一致。

\SolAnswer 投影系数$a=2$，投影$\boldsymbol p=(2,2)^{\mathsf T}$，残差$\boldsymbol r=(1,-1)^{\mathsf T}$，距离为$\sqrt2$。
\end{solution}

\subsection*{主特征方向的严格伪代码}
\phantomsection\label{schema:V1-C02:CH-19}
\phantomsection\label{schema:V1-C02:CH-20}
输入为方阵$A\in\R^{n\times n}$与非零初始向量，输出为最大模特征值及其单位特征向量的近似与残差；若最大模特征值为负，它不等于Rayleigh商最大值。方向收敛还要求最大模特征值唯一且初始向量在该特征方向上的分量非零。
\phantomsection\label{schema:V1-C02:CH-21}
参数$T$是最大迭代次数，$\varepsilon>0$是残差容差；符号$q_t$表示第$t$次单位向量。
\phantomsection\label{schema:V1-C02:CH-22}
初始化使用$q_0=v_0/\norm{v_0}_2$，要求$v_0\ne0$。
\phantomsection\label{schema:V1-C02:CH-23}
更新依次计算$z_t=Aq_{t-1}$、$q_t=z_t/\norm{z_t}_2$和$\lambda_t=q_t^{\mathsf T}Aq_t$。
\phantomsection\label{schema:V1-C02:CH-24}
停止条件为特征残差$\norm{Aq_t-\lambda_tq_t}_2\le\varepsilon$或$t=T$。
\phantomsection\label{schema:V1-C02:CH-25}
本章只在实对称矩阵上给出无条件数遗漏的速率结论。设
$A=A^{\mathsf T}$有标准正交特征向量$q_1,\ldots,q_n$，特征值按
$|\lambda_1|>|\lambda_2|\ge\cdots\ge|\lambda_n|$排列，并写
$q_0=\sum_i c_iq_i$且$c_1\ne0$。若迭代未遇到$Aq_{t-1}=0$，则归一化不改变方向，并且
\[
\tan\angle\!\bigl(q_t,\operatorname{span}(q_1)\bigr)
=\frac{\bigl(\sum_{i=2}^n c_i^2|\lambda_i/\lambda_1|^{2t}\bigr)^{1/2}}{|c_1|}
\le \frac{\bigl(\sum_{i=2}^n c_i^2\bigr)^{1/2}}{|c_1|}
\left|\frac{\lambda_2}{\lambda_1}\right|^t.
\]
一般可对角化但非正规矩阵还会出现特征向量基的条件数，复特征值与Jordan块也需另行处理，不能直接套用上述常数为一的方向界。

% KNOWLEDGE-PLAN: KN-V1-C02-ALGORITHM_IDEA-001 L1
\paragraph{读前自检：归一化幂法：执行契约。}归一化幂法以方阵$A$、非零$v_0$、$\varepsilon>0$和$T\in\mathbb N_0$为输入；请从合法$\widehat q$计算$z=A\widehat q$、$q^+=z/\lVert z\rVert_2$、$\lambda^+$与$r^+$，只在候选均有限时提交，并核对$r^+\le\varepsilon$触发$\mathtt{converged}$，否则到$T$轮触发$\mathtt{budget\_stop}$。\par
% KNOWLEDGE-PLAN: KN-V1-C02-ALGORITHM_IDEA-002 L2
\begin{keypointbox}[title={归一化幂法：五步阅读}]
\textbf{输入。}写出数据对象、固定参数与必须成立的条件。\par
\textbf{初始化。}标明主状态、计数器和第一个合法状态。\par
\textbf{核心更新。}只手算一次从旧状态到候选状态的更新，并指出何时提交。\par
\textbf{数学停止。}写出能直接核验的停止证书；预算耗尽不等同于收敛。\par
\textbf{输出。}列出返回对象、实际计数、中心状态和用于复核的诊断。
\end{keypointbox}

\begin{AlgorithmContract}{
  input={方阵$A$、非零初始向量$v_0$、残差容差$\varepsilon$与非负轮预算$T$},
  output={最后合法特征对近似$(\widehat\lambda,\widehat q)$、残差$\widehat r$、实际提交轮数$t_{\rm done}$、矩阵向量乘法数$m_{\rm done}$、状态与最终诊断},
  preconditions={$A\in\mathbb R^{n\times n}$且元素有限，$v_0$有限非零，$\varepsilon>0$，$T\in\mathbb N_0$},
  initialization={归一化$v_0$得到$\widehat q$，计算初始Rayleigh商和残差，置$t_{\rm done}=0$并记录实际乘法数},
  loop={在$t_{\rm done}<T$且初始或当前残差未达标时，形成下一单位方向、Rayleigh商和残差候选},
  update={候选范数、商与残差全部有限后才原子提交$(\widehat\lambda,\widehat q,\widehat r)$并增加$t_{\rm done}$},
  domain={每次归一化分母严格为正；单位向量、Rayleigh商与特征残差均为有限实数},
  failure={输入非法返回$\mathtt{invalid\_input}$；零归一化分母或非有限乘法/残差返回$\mathtt{numerical\_failure}$并保留上一合法特征对},
  stop={残差不超过$\varepsilon$、轮预算耗尽或首次失败时停止},
  budget={至多原子提交$T$轮；初始化和每轮各按实际发生的矩阵向量乘法计数},
  status={$\mathtt{converged}$、$\mathtt{budget\_stop}$、$\mathtt{invalid\_input}$、$\mathtt{numerical\_failure}$},
  iterations={$t_{\rm done}$为实际提交轮数，$m_{\rm done}$为实际矩阵向量乘法数，失败轮不计入前者},
  diagnostics={最终残差、向量范数、Rayleigh商、预算余量以及首个失败轮与计算阶段},
  complexity={稠密情形$O(m_{\rm done}n^2)$时间，稀疏情形$O(m_{\rm done}\operatorname{nnz}(A))$时间，额外空间$O(n)$}
}
\begin{algorithm}[H]
\caption{归一化幂法}\label{alg:V1-C02-power}
\KwIn{$A\in\R^{n\times n}$，$v_0\ne0$，$\varepsilon>0$，$T\in\mathbb N_0$}
\KwOut{最后合法$(\widehat\lambda,\widehat q,\widehat r)$；$t_{\rm done},m_{\rm done}$；$\mathtt{status}$与诊断}
若维数、有限性、$v_0\ne0$、容差或预算非法，返回空结果、计数$0$、$\mathtt{invalid\_input}$及字段诊断\;
置$\widehat q\leftarrow v_0/\|v_0\|_2,t_{\rm done}\leftarrow0,m_{\rm done}\leftarrow0$；计算$y\leftarrow A\widehat q$并增加$m_{\rm done}$\;
若$y$非有限，则返回$\widehat q$、$\mathtt{numerical\_failure}$及初始化乘法诊断；否则置$\widehat\lambda\leftarrow\widehat q^{\mathsf T}y,\widehat r\leftarrow\|y-\widehat\lambda\widehat q\|_2$\;
若$\widehat\lambda$或$\widehat r$非有限，返回最后合法方向、$\mathtt{numerical\_failure}$及初始残差诊断；若$\widehat r\le\varepsilon$，返回当前结果、$\mathtt{converged}$及零轮残差证书\;
\While{$t_{\rm done}<T$}{
  计算$z\leftarrow A\widehat q$并增加$m_{\rm done}$；若$z$非有限或$\|z\|_2=0$，返回上一合法结果、$\mathtt{numerical\_failure}$及失败轮诊断\;
  形成$q^+\leftarrow z/\|z\|_2$，计算$y^+\leftarrow Aq^+$并增加$m_{\rm done}$，再形成$\lambda^+=(q^+)^{\mathsf T}y^+$与$r^+=\|y^+-\lambda^+q^+\|_2$\;
  若候选任一量非有限，则返回上一合法结果、$\mathtt{numerical\_failure}$及候选阶段诊断\;
  原子提交$(\widehat q,\widehat\lambda,\widehat r)\leftarrow(q^+,\lambda^+,r^+)$并增加$t_{\rm done}$\;
  若$\widehat r\le\varepsilon$，返回最后合法结果、$\mathtt{converged}$及残差证书\;
}
返回最后合法结果、$\mathtt{budget\_stop}$及“预算耗尽、残差未达标”最终诊断\;
\end{algorithm}
\end{AlgorithmContract}

\SLReviewSection{内积、范数、距离与夹角}{sec:V1-C02-S02}

$\inner{\boldsymbol{x},\boldsymbol{y}}=\sum_i x_i y_i$是标量，满足对称性、双线性和正定性

\inner{\boldsymbol{x},\boldsymbol{x}}=0

最后一个“当且仅当”把内积与仅半正定的双线性型区分开；也正因为它，$\norm{\boldsymbol{x}}_2=\sqrt{\inner{\boldsymbol{x},\boldsymbol{x}}}$满足$\norm{\boldsymbol{x}}_2=0\Leftrightarrow\boldsymbol{x}=\boldsymbol0$。

$S^{\perp}=\{\boldsymbol{x}:\boldsymbol{x}^{\mathsf T}\boldsymbol{s}=0,\ \forall\boldsymbol{s}\in S\}$也是子空间。
% KNOWLEDGE-PLAN: KN-V1-C02-PREREQUISITE-020 L1
\paragraph{读前自检：矩阵的基本子空间。}对$A\in\R^{m\times n}$，请核对定义域中的$R(A^{\mathsf T})\perp N(A)$与输出域中的$R(A)\perp N(A^{\mathsf T})$，再用两条秩--零度等式验证两对正交补维数闭合。\par

\knowledgeanchor{SLM-D-06-001}{矩阵的基本子空间}
四个空间成两对正交补：定义域中是行空间与零空间，输出空间中是列空间与左零空间。
% KNOWLEDGE-PLAN: KN-V1-C02-THEOREM-003 L1
\paragraph{读前自检：定理D 1。}定理D.1断言$N(A)=R(A^{\mathsf T})^{\perp}$且$N(A^{\mathsf T})=R(A)^{\perp}$；请把“$\boldsymbol{x}$与$A$的每一行正交”改写成$A\boldsymbol{x}=0$，再对$A^{\mathsf T}$重复同一步，并核对$N(A)=R(A^{\mathsf T})^\perp$及其转置版本同时成立。\par

\knowledgeanchor{SLM-D-01-002}{定理D.1}
具体关系为
\[
N(A)=R(A^{\mathsf T})^{\perp},\qquad
N(A^{\mathsf T})=R(A)^{\perp}.
\]
证明第一式时，$\boldsymbol{x}$与$A$每一行正交等价于$A\boldsymbol{x}=0$；第二式对$A^{\mathsf T}$应用同一论证。

% KNOWLEDGE-PLAN: KN-V1-C02-CONCEPT-011 L1
\paragraph{读前自检：正交投影。}在“若$Q$的列标准正交”成立时处理线性代数基础中的正交投影：请独立化简$P=QQ^{\mathsf T}$的两端，并以到$R(Q)$的投影矩阵为$P=QQ^{\mathsf T}$判定结果。\par
% KNOWLEDGE-PLAN: KN-V1-C02-CONCEPT-012 L1
\paragraph{读前自检：点到超平面的距离。}点$\boldsymbol x_0$到超平面$\boldsymbol w^{\mathsf T}\boldsymbol x+b=0$的距离要求$\boldsymbol w\ne0$；请把$\boldsymbol x_0$沿单位法向量投影，核对距离为$|\boldsymbol w^{\mathsf T}\boldsymbol x_0+b|/\lVert\boldsymbol w\rVert_2$。\par

\knowledgeanchor{PRE-LA-014}{正交投影}
若$Q$的列标准正交，则到$R(Q)$的投影矩阵为$P=QQ^{\mathsf T}$，满足$P^{\mathsf T}=P$和$P^2=P$。
\knowledgeanchor{PRE-LA-025}{点到超平面的距离}
点$\boldsymbol{x}_0$到$\boldsymbol{w}^{\mathsf T}\boldsymbol{x}+b=0$的距离为$|\boldsymbol{w}^{\mathsf T}\boldsymbol{x}_0+b|/\norm{\boldsymbol{w}}_2$，前提是$\boldsymbol{w}\ne0$。

\paragraph{读前自检：特征值与特征向量。}围绕特征值与特征向量，先采用条件“若$A\boldsymbol{q}=\lambda\boldsymbol{q}$且$\boldsymbol{q}\ne0$”，再标出$A\boldsymbol{q}=\lambda\boldsymbol{q}$两侧的形状并完成一次乘法或正交检查；最后验证$\lambda$是特征值、$\boldsymbol{q}$是特征向量。\par

% KNOWLEDGE-PLAN: KN-V1-C02-PREREQUISITE-023 L1
\paragraph{读前自检：正定矩阵与半正定矩阵。}把正定矩阵与半正定矩阵放在“对称矩阵$A$若对所有非零$\boldsymbol{x}$有$\boldsymbol{x}^{\mathsf T}A\boldsymbol{x}>0$则正定”下考察：标出$\boldsymbol{x}^{\mathsf T}A\boldsymbol{x}>0$两侧的形状并完成一次乘法或正交检查，并检查若对所有$\boldsymbol{x}$非负则半正定。\par
% KNOWLEDGE-PLAN: KN-V1-C02-CONCEPT-013 L1

\knowledgeanchor{PRE-LA-016}{特征值与特征向量}

\knowledgeanchor{PRE-LA-018}{正定矩阵与半正定矩阵}

\knowledgeanchor{PRE-LA-019}{二次型}
$q(\boldsymbol{x})=\boldsymbol{x}^{\mathsf T}A\boldsymbol{x}$是二次型，只有$A$的对称部分$(A+A^{\mathsf T})/2$影响其值。
\knowledgeanchor{PRE-LA-020}{矩阵的迹}
$\tr(A)=\sum_iA_{ii}$；对维数相容矩阵有$\tr(AB)=\tr(BA)$，并且迹等于特征值按代数重数之和。

实对称矩阵可写为$A=Q\Lambda Q^{\mathsf T}$，其中$Q$正交，$\Lambda$为实对角矩阵；二次型因此分解为各特征方向坐标平方的加权和。

\end{aligned}
\]

AB&=\begin{bmatrix}1\cdot1+1\cdot1&1\cdot0+1\cdot1\\
0\cdot1+1\cdot1&0\cdot0+1\cdot1\end{bmatrix}

BA&=\begin{bmatrix}1&1\\1&2\end{bmatrix},
&AB-BA&=\begin{bmatrix}1&0\\0&-1\end{bmatrix}\ne0.
\end{aligned}
\]
因此该例直接说明一般矩阵乘法不满足交换律。

\phantomsection\label{sol:V1-C02-S05-02}
\begin{chapterexercisesolution}{ex:V1-C02-S05-02}
给定$\boldsymbol q=(1,0)^{\mathsf T}\in\R^2$且$\boldsymbol q^{\mathsf T}\boldsymbol q=1$，故一维投影矩阵无需额外归一化：
\[
\begin{aligned}
P&=\boldsymbol q\boldsymbol q^{\mathsf T}
=\begin{bmatrix}1&0\\0&0\end{bmatrix}\in\R^{2\times2},\\
P\begin{bmatrix}2\\3\end{bmatrix}
&=\begin{bmatrix}2\\0\end{bmatrix},
&\boldsymbol r&=\begin{bmatrix}2\\3\end{bmatrix}
-\begin{bmatrix}2\\0\end{bmatrix}=\begin{bmatrix}0\\3\end{bmatrix},\\
\boldsymbol q^{\mathsf T}\boldsymbol r&=0,
&P^2&=P.
\end{aligned}
\]
因此$P$确为到$x$轴的正交投影，向量$(2,3)^{\mathsf T}$的投影为$(2,0)^{\mathsf T}$。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[点到直线距离]\label{ex:V1-C02-S05-03}
求点$(0,0)^{\mathsf T}$到直线$3x_1+4x_2-10=0$的距离。
\end{exercise}
\phantomsection\label{sol:V1-C02-S05-03}
\begin{chapterexercisesolution}{ex:V1-C02-S05-03}
直线法向量$\boldsymbol w=(3,4)^{\mathsf T}\ne0$，故点到超平面的距离公式分母非零。对$\boldsymbol x_0=(0,0)^{\mathsf T}$，
\[
\begin{aligned}
\norm{\boldsymbol w}_2&=\sqrt{3^2+4^2}=5,\\
|\boldsymbol w^{\mathsf T}\boldsymbol x_0-10|
&=|3\cdot0+4\cdot0-10|=10,\\
\operatorname{dist}(\boldsymbol x_0,H)&=\frac{10}{5}=2\ge0.
\end{aligned}
\]
所以原点到该直线的距离为$2$。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[特征对]\label{ex:V1-C02-S06-01}
验证$(1,1)^{\mathsf T}$是$A=\begin{bmatrix}2&1\\1&2\end{bmatrix}$的特征向量并求特征值。
\end{exercise}
\phantomsection\label{sol:V1-C02-S06-01}
\begin{chapterexercisesolution}{ex:V1-C02-S06-01}
对象为$A\in\R^{2\times2}$和非零候选$\boldsymbol q=(1,1)^{\mathsf T}$。按特征对定义核对
\[
\begin{aligned}
A\boldsymbol q
&=\begin{bmatrix}2&1\\1&2\end{bmatrix}
\begin{bmatrix}1\\1\end{bmatrix}
=\begin{bmatrix}3\\3\end{bmatrix}\\
&=3\begin{bmatrix}1\\1\end{bmatrix}=3\boldsymbol q.
\end{aligned}
\]
因此$\boldsymbol q$是特征向量，对应特征值$\lambda=3$。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[正定判断]\label{ex:V1-C02-S06-02}
判断$A=\operatorname{diag}(2,0)$是正定还是半正定。
\end{exercise}
\phantomsection\label{sol:V1-C02-S06-02}
\begin{chapterexercisesolution}{ex:V1-C02-S06-02}
矩阵$A=\operatorname{diag}(2,0)$实对称。对任意$\boldsymbol x=(x_1,x_2)^{\mathsf T}\in\R^2$，
\[
\begin{aligned}
\boldsymbol x^{\mathsf T}A\boldsymbol x
&=\begin{bmatrix}x_1&x_2\end{bmatrix}
\begin{bmatrix}2x_1\\0\end{bmatrix}=2x_1^2\ge0,\\
\boldsymbol e_2^{\mathsf T}A\boldsymbol e_2
&=0,\qquad \boldsymbol e_2=(0,1)^{\mathsf T}\ne\boldsymbol0.
\end{aligned}
\]
故$A$半正定，但因存在非零零方向而不正定。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[迹与特征值]\label{ex:V1-C02-S06-03}
已知二阶矩阵特征值为$5$和$-1$，求其迹，并判断它能否为正定对称矩阵。
\end{exercise}
\phantomsection\label{sol:V1-C02-S06-03}
\begin{chapterexercisesolution}{ex:V1-C02-S06-03}
对象是实二阶矩阵，给定特征值按代数重数为$\lambda_1=5$、$\lambda_2=-1$。于是
\[
\begin{aligned}
\tr(A)&=\lambda_1+\lambda_2=5-1=4,\\
A=A^{\mathsf T},\ A\succ0
&\Longrightarrow \lambda_1>0\ \text{且}\ \lambda_2>0,\\
\lambda_2=-1<0&\Longrightarrow A\text{不可能是正定对称矩阵}.
\end{aligned}
\]
迹为正只约束特征值之和，不能排除负特征值。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[谱重构]\label{ex:V1-C02-S07-01}
设$Q=I_2$、$\Lambda=\operatorname{diag}(3,1)$，写出$Q\Lambda Q^{\mathsf T}$并解释两个主轴。
\end{exercise}
\phantomsection\label{sol:V1-C02-S07-01}
\begin{chapterexercisesolution}{ex:V1-C02-S07-01}
对象为$Q=I_2\in\R^{2\times2}$与$\Lambda=\operatorname{diag}(3,1)$。谱重构及主轴作用为
\[
\begin{aligned}
Q\Lambda Q^{\mathsf T}
&=I_2\begin{bmatrix}3&0\\0&1\end{bmatrix}I_2
=\begin{bmatrix}3&0\\0&1\end{bmatrix},\\
A\boldsymbol e_1&=3\boldsymbol e_1,
&A\boldsymbol e_2&=1\boldsymbol e_2.
\end{aligned}
\]
故两条主轴就是标准坐标轴，第一轴伸缩三倍，第二轴保持不变。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[奇异值]\label{ex:V1-C02-S07-02}
求$A=\operatorname{diag}(4,2)$的奇异值。
\end{exercise}
\phantomsection\label{sol:V1-C02-S07-02}
\begin{chapterexercisesolution}{ex:V1-C02-S07-02}
对象为$A=\operatorname{diag}(4,2)\in\R^{2\times2}$；奇异值是$A^{\mathsf T}A$特征值的非负平方根：
\[
\begin{aligned}
A^{\mathsf T}A
&=\begin{bmatrix}4&0\\0&2\end{bmatrix}^{\mathsf T}
\begin{bmatrix}4&0\\0&2\end{bmatrix}
=\begin{bmatrix}16&0\\0&4\end{bmatrix},\\
\lambda(A^{\mathsf T}A)&\in\{16,4\},\\
(\sigma_1,\sigma_2)&=(\sqrt{16},\sqrt4)=(4,2).
\end{aligned}
\]
因此按非增次序排列的奇异值为$4$和$2$。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[幂法一步]\label{ex:V1-C02-S07-03}
对$A=\operatorname{diag}(3,1)$和$q_0=(1,1)^{\mathsf T}/\sqrt2$执行一次归一化幂法。
\end{exercise}
\phantomsection\label{sol:V1-C02-S07-03}
\begin{chapterexercisesolution}{ex:V1-C02-S07-03}
输入满足$A\in\R^{2\times2}$、$\norm{q_0}_2=1$，且$z_1=Aq_0\ne0$，所以本轮归一化合法。一次完整更新为
\[
\begin{aligned}
z_1&=Aq_0
=\frac1{\sqrt2}\begin{bmatrix}3&0\\0&1\end{bmatrix}
\begin{bmatrix}1\\1\end{bmatrix}
=\frac1{\sqrt2}\begin{bmatrix}3\\1\end{bmatrix},\\
\norm{z_1}_2&=\sqrt{\frac{9+1}{2}}=\sqrt5,
&q_1&=\frac{z_1}{\norm{z_1}_2}=\frac1{\sqrt{10}}\begin{bmatrix}3\\1\end{bmatrix},\\
\lambda_1&=q_1^{\mathsf T}Aq_1=\frac{27+1}{10}=\frac{14}{5},
&\norm{Aq_1-\lambda_1q_1}_2&=\frac35.
\end{aligned}
\]
方向系数比由$1:1$变为$3:1$，主特征方向的相对权重增大。
\end{chapterexercisesolution}

\Needspace{6\baselineskip}
\begin{exercise}[章末综合]\label{ex:V1-C02-CH-01}
综合题：设$A=\begin{bmatrix}1&1\\1&1\\0&0\end{bmatrix}$。求秩、零空间一组基、列空间一组基，并验证$N(A)=R(A^{\mathsf T})^{\perp}$。
\end{exercise}
\phantomsection\label{sol:V1-C02-CH-01}
\begin{chapterexercisesolution}{ex:V1-C02-CH-01}
对象为$A:\R^2\to\R^3$。两列相同且非零，故依次得到
\[
\begin{aligned}
\rank(A)&=1,\\
R(A)&=\operatorname{span}\left\{\begin{bmatrix}1\\1\\0\end{bmatrix}\right\}\subseteq\R^3,\\
A\begin{bmatrix}x_1\\x_2\end{bmatrix}=\boldsymbol0
&\Longleftrightarrow x_1+x_2=0,\\
N(A)&=\operatorname{span}\left\{\begin{bmatrix}-1\\1\end{bmatrix}\right\}\subseteq\R^2,\\
R(A^{\mathsf T})
&=\operatorname{span}\left\{\begin{bmatrix}1\\1\end{bmatrix}\right\},\\
R(A^{\mathsf T})^\perp
&=\left\{\begin{bmatrix}z_1\\z_2\end{bmatrix}:z_1+z_2=0\right\}\\
&=\operatorname{span}\left\{\begin{bmatrix}-1\\1\end{bmatrix}\right\}=N(A).
\end{aligned}
\]
因此$N(A)=R(A^{\mathsf T})^\perp$在本例中成立，同时$\rank(A)=1$与各空间维数相符。
\end{chapterexercisesolution}
